```java
@Component
public class FraudDetectionStreamsApp {
    
    private static final String TRANSACTIONS_TOPIC = "transactions";
    private static final String FRAUD_ALERTS_TOPIC = "fraud-alerts";
    private static final String CUSTOMER_PROFILES_TOPIC = "customer-profiles";
    
    public Topology buildTopology() {
        StreamsBuilder builder = new StreamsBuilder();
        
        // Transaction input stream
        KStream<String, Transaction> transactions = builder
            .stream(TRANSACTIONS_TOPIC, Consumed.with(Serdes.String(), transactionSerde));
        
        // Customer profiles table
        KTable<String, CustomerProfile> customerProfiles = builder
            .table(CUSTOMER_PROFILES_TOPIC, Consumed.with(Serdes.String(), customerProfileSerde));
        
        // Fraud detection pipeline
        KStream<String, FraudAlert> fraudAlerts = transactions
            .peek((key, txn) -> log.debug("Processing transaction: {}", txn.getId()))
            
            // Join with customer profile for enrichment
            .leftJoin(customerProfiles, this::enrichTransaction,
                Joined.with(Serdes.String(), transactionSerde, customerProfileSerde))
            
            // Apply fraud detection rules
            .flatMapValues(this::detectFraud)
            .filter((key, alert) -> alert != null);
        
        // Velocity-based fraud detection (windowed aggregations)
        KStream<String, FraudAlert> velocityAlerts = transactions
            .groupBy((key, txn) -> txn.getCustomerId())
            .windowedBy(TimeWindows.of(Duration.ofMinutes(5)).grace(Duration.ofSeconds(30)))
            .aggregate(
                TransactionVelocity::new,
                this::aggregateTransactionVelocity,
                Materialized.<String, TransactionVelocity, WindowStore<Bytes, byte[]>>as("velocity-store")
                    .withKeySerde(Serdes.String())
                    .withValueSerde(velocitySerde)
            )
            .toStream()
            .flatMapValues(this::checkVelocityFraud)
            .filter((key, alert) -> alert != null);
        
        // Combine all fraud detection streams
        KStream<String, FraudAlert> allFraudAlerts = fraudAlerts.merge(velocityAlerts);
        
        // Apply fraud scoring and filtering
        KStream<String, FraudAlert> finalAlerts = allFraudAlerts
            .mapValues(this::calculateFraudScore)
            .filter((key, alert) -> alert.getScore() >= getFraudThreshold())
            .peek((key, alert) -> log.info("Fraud detected: {}", alert));
        
        // Output to alerts topic
        finalAlerts.to(FRAUD_ALERTS_TOPIC, 
            Produced.with(Serdes.String(), fraudAlertSerde));
        
        return builder.build();
    }
    
    private EnrichedTransaction enrichTransaction(Transaction transaction, CustomerProfile profile) {
        return EnrichedTransaction.builder()
            .transaction(transaction)
            .customerProfile(profile)
            .riskLevel(profile != null ? profile.getRiskLevel() : RiskLevel.MEDIUM)
            .build();
    }
    
    private List<FraudAlert> detectFraud(EnrichedTransaction enriched) {
        List<FraudAlert> alerts = new ArrayList<>();
        Transaction txn = enriched.getTransaction();
        CustomerProfile profile = enriched.getCustomerProfile();
        
        // Rule 1: Amount-based anomaly detection
        if (profile != null) {
            double avgAmount = profile.getAverageTransactionAmount();
            if (txn.getAmount() > avgAmount * 10) { // 10x average
                alerts.add(FraudAlert.builder()
                    .transactionId(txn.getId())
                    .customerId(txn.getCustomerId())
                    .fraudType(FraudType.AMOUNT_ANOMALY)
                    .description(String.format("Amount %.2f is 10x average %.2f", 
                        txn.getAmount(), avgAmount))
                    .severity(Severity.HIGH)
                    .timestamp(txn.getTimestamp())
                    .build());
            }
        }
        
        // Rule 2: Geographic anomaly
        if (profile != null && !isValidGeography(txn, profile)) {
            alerts.add(FraudAlert.builder()
                .transactionId(txn.getId())
                .customerId(txn.getCustomerId())
                .fraudType(FraudType.GEOGRAPHIC_ANOMALY)
                .description("Transaction from unusual location")
                .severity(Severity.MEDIUM)
                .timestamp(txn.getTimestamp())
                .build());
        }
        
        // Rule 3: Blacklist check
        if (isBlacklistedMerchant(txn.getMerchantId()) || 
            isBlacklistedLocation(txn.getLocation())) {
            alerts.add(FraudAlert.builder()
                .transactionId(txn.getId())
                .customerId(txn.getCustomerId())
                .fraudType(FraudType.BLACKLIST_MATCH)
                .description("Blacklisted merchant or location")
                .severity(Severity.HIGH)
                .timestamp(txn.getTimestamp())
                .build());
        }
        
        return alerts;
    }
    
    private TransactionVelocity aggregateTransactionVelocity(String customerId, 
                                                           Transaction txn, 
                                                           TransactionVelocity velocity) {
        return velocity.addTransaction(txn);
    }
    
    private List<FraudAlert> checkVelocityFraud(Windowed<String> windowedKey, 
                                              TransactionVelocity velocity) {
        String customerId = windowedKey.key();
        
        // Check if velocity exceeds thresholds
        if (velocity.getTransactionCount() > 20 || // More than 20 transactions in 5 minutes
            velocity.getTotalAmount() > 10000 ||   // More than $10k in 5 minutes
            velocity.getUniqueLocations() > 5) {   // More than 5 different locations
            
            return Collections.singletonList(
                FraudAlert.builder()
                    .customerId(customerId)
                    .fraudType(FraudType.VELOCITY_ANOMALY)
                    .description(String.format(
                        "High velocity: %d transactions, $%.2f total, %d locations in 5 minutes",
                        velocity.getTransactionCount(),
                        velocity.getTotalAmount(),
                        velocity.getUniqueLocations()))
                    .severity(Severity.HIGH)
                    .timestamp(Instant.now())
                    .build());
        }
        
        return Collections.emptyList();
    }
}
```

```java
public class RecommendationStreamsTopology {
    
    public Topology buildTopology() {
        StreamsBuilder builder = new StreamsBuilder();
        
        // Input streams
        KStream<String, UserInteraction> interactions = builder.stream("user-interactions");
        KStream<String, Product> productUpdates = builder.stream("product-updates");
        
        // Reference data tables
        KTable<String, UserProfile> userProfiles = builder.table("user-profiles");
        KTable<String, Product> productCatalog = builder.table("product-catalog");
        
        // Real-time user behavior aggregation
        KTable<String, UserBehaviorProfile> userBehavior = interactions
            .groupByKey()
            .windowedBy(TimeWindows.of(Duration.ofHours(24)).grace(Duration.ofMinutes(5)))
            .aggregate(
                UserBehaviorProfile::new,
                this::aggregateUserBehavior,
                Materialized.as("user-behavior-store")
            )
            .suppress(Suppressed.untilWindowCloses(Suppressed.BufferConfig.unbounded()));
        
        // Product popularity calculation
        KTable<String, ProductPopularity> productPopularity = interactions
            .filter((key, interaction) -> "purchase".equals(interaction.getType()))
            .groupBy((key, interaction) -> interaction.getProductId())
            .windowedBy(TimeWindows.of(Duration.ofHours(6)))
            .count(Materialized.as("product-popularity-store"))
            .mapValues(ProductPopularity::fromCount);
        
        // Session-based recommendations
        KStream<String, Recommendation> sessionRecommendations = interactions
            .groupByKey()
            .windowedBy(SessionWindows.with(Duration.ofMinutes(30)).grace(Duration.ofMinutes(5)))
            .aggregate(
                UserSession::new,
                this::aggregateSession,
                this::mergeSession,
                Materialized.as("user-sessions")
            )
            .toStream()
            .flatMapValues(this::generateSessionRecommendations);
        
        // Join enriched recommendations
        KStream<String, EnrichedRecommendation> enrichedRecommendations = sessionRecommendations
            .leftJoin(userProfiles, this::enrichWithProfile,
                Joined.with(Serdes.String(), recommendationSerde, userProfileSerde))
            .leftJoin(productCatalog, this::enrichWithProductData,
                Joined.with(Serdes.String(), enrichedRecommendationSerde, productSerde));
        
        // A/B testing branch
        KStream<String, Recommendation>[] testingBranches = enrichedRecommendations
            .mapValues(this::finalizeRecommendation)
            .branch(
                (key, rec) -> isInTestGroup(key, "algorithm_v2"),
                (key, rec) -> true  // Control group
            );
        
        // Different algorithms for A/B testing
        testingBranches[0]
            .mapValues(rec -> applyAlgorithmV2(rec))
            .to("recommendations-v2");
            
        testingBranches[1]
            .mapValues(rec -> applyAlgorithmV1(rec))
            .to("recommendations-v1");
        
        return builder.build();
    }
    
    private UserBehaviorProfile aggregateUserBehavior(String userId, 
                                                     UserInteraction interaction, 
                                                     UserBehaviorProfile profile) {
        return profile.addInteraction(interaction);
    }
    
    private UserSession aggregateSession(String userId, 
                                       UserInteraction interaction, 
                                       UserSession session) {
        return session.addInteraction(interaction);
    }
    
    private UserSession mergeSession(String userId, 
                                   UserSession session1, 
                                   UserSession session2) {
        return session1.merge(session2);
    }
    
    private List<Recommendation> generateSessionRecommendations(Windowed<String> windowedKey, 
                                                              UserSession session) {
        String userId = windowedKey.key();
        
        // Collaborative filtering based on session patterns
        List<String> viewedCategories = session.getViewedCategories();
        List<String> viewedProducts = session.getViewedProducts();
        
        // Generate recommendations using multiple strategies
        List<Recommendation> recommendations = new ArrayList<>();
        
        // 1. Category-based recommendations
        recommendations.addAll(getCategoryBasedRecommendations(userId, viewedCategories));
        
        // 2. Similar products recommendations
        recommendations.addAll(getSimilarProductRecommendations(userId, viewedProducts));
        
        // 3. Trending in category recommendations  
        recommendations.addAll(getTrendingRecommendations(userId, viewedCategories));
        
        // Score and rank recommendations
        return recommendations.stream()
            .sorted((r1, r2) -> Double.compare(r2.getScore(), r1.getScore()))
            .limit(10)
            .collect(Collectors.toList());
    }
}
```

#### Cold Start Handling Strategy

```java
@Component
public class ColdStartRecommendationEngine {
    
    public List<Recommendation> generateColdStartRecommendations(String userId, int limit) {
        // Strategy 1: Popular products globally
        List<Recommendation> popularProducts = getGlobalPopularProducts(limit / 2);
        
        // Strategy 2: New user onboarding recommendations
        List<Recommendation> onboardingProducts = getOnboardingRecommendations(limit / 2);
        
        // Combine and randomize to avoid bias
        List<Recommendation> combined = new ArrayList<>();
        combined.addAll(popularProducts);
        combined.addAll(onboardingProducts);
        
        Collections.shuffle(combined);
        
        return combined.stream()
            .limit(limit)
            .map(rec -> rec.withExplanation("Popular products to get you started"))
            .collect(Collectors.toList());
    }
    
    private List<Recommendation> getGlobalPopularProducts(int limit) {
        // Query product popularity aggregation
        ReadOnlyWindowStore<String, ProductPopularity> popularityStore = 
            kafkaStreams.store(
                StoreQueryParameters.fromNameAndType("product-popularity-store", 
                    QueryableStoreTypes.windowStore())
            );
        
        Instant now = Instant.now();
        Instant sixHoursAgo = now.minus(Duration.ofHours(6));
        
        Map<String, Long> productCounts = new HashMap<>();
        
        try (KeyValueIterator<Windowed<String>, Long> iterator = 
             popularityStore.fetchAll(sixHoursAgo, now)) {
            
            while (iterator.hasNext()) {
                KeyValue<Windowed<String>, Long> entry = iterator.next();
                String productId = entry.key.key();
                Long count = entry.value;
                
                productCounts.merge(productId, count, Long::sum);
            }
        }
        
        return productCounts.entrySet().stream()
            .sorted(Map.Entry.<String, Long>comparingByValue().reversed())
            .limit(limit)
            .map(entry -> Recommendation.builder()
                .productId(entry.getKey())
                .score(entry.getValue() / 1000.0)  // Normalize score
                .reason("Popular product")
                .build())
            .collect(Collectors.toList());
    }
}
```

```java
@Component
public class OrderProcessingSaga {
    
    @KafkaListener(topics = "order-events")
    public void handleOrderEvent(ConsumerRecord<String, OrderEvent> record) {
        OrderEvent event = record.value();
        String orderId = event.getOrderId();
        
        switch (event) {
            case OrderPlaced orderPlaced -> handleOrderPlaced(orderPlaced);
            case OrderAccepted orderAccepted -> handleOrderAccepted(orderAccepted);
            case OrderReady orderReady -> handleOrderReady(orderReady);
            case OrderPickedUp orderPickedUp -> handleOrderPickedUp(orderPickedUp);
            case OrderCancelled orderCancelled -> handleOrderCancelled(orderCancelled);
            default -> log.warn("Unhandled event type: {}", event.getClass().getSimpleName());

    private void handleOrderPlaced(OrderPlaced event) {

        InventoryReservationCommand reserveCmd = InventoryReservationCommand.builder()
            .orderId(event.getOrderId())
            .restaurantId(event.getRestaurantId())
            .items(event.getItems())
            .build();

        PaymentAuthCommand paymentCmd = PaymentAuthCommand.builder()
            .orderId(event.getOrderId())
            .customerId(event.getCustomerId())
            .amount(event.getTotalAmount())
            .paymentMethod(event.getPaymentMethod())
            .build();
        
        commandBus.send("payment-commands", event.getOrderId(), paymentCmd);

            .orderId(event.getOrderId())
            .restaurantId(event.getRestaurantId())
            .orderDetails(event.getItems())
            .customerInfo(event.getCustomerId())
            .build();
        
        commandBus.send("restaurant-commands", event.getRestaurantId(), notifyCmd);
    }
    
    private void handleOrderReady(OrderReady event) {
        // Trigger driver assignment
        DriverAssignmentCommand assignCmd = DriverAssignmentCommand.builder()
            .orderId(event.getOrderId())
            .restaurantLocation(getRestaurantLocation(event.getRestaurantId()))
            .deliveryAddress(getOrderDeliveryAddress(event.getOrderId()))
            .priority(calculateDeliveryPriority(event))
            .build();
        
        commandBus.send("driver-assignment-commands", event.getOrderId(), assignCmd);

            .customerId(event.getCustomerId())
            .orderId(event.getOrderId())
            .message("Your order is ready and a driver is being assigned!")
            .notificationType(NotificationType.ORDER_UPDATE)
            .build();
        
        commandBus.send("notification-commands", event.getCustomerId(), customerNotify);
    }

            .orderId(event.getOrderId())
            .restaurantId(event.getRestaurantId())
            .build();

        // 2. Process refund

            .orderId(event.getOrderId())
            .refundReason(event.getCancellationReason())
            .build();
        
        commandBus.send("payment-commands", event.getOrderId(), refundCmd);
        
        // 3. Update driver if assigned
        if (event.getDriverId() != null) {
            DriverReleaseCommand driverRelease = DriverReleaseCommand.builder()
                .driverId(event.getDriverId())
                .orderId(event.getOrderId())
                .build();
            
            commandBus.send("driver-commands", event.getDriverId(), driverRelease);
        }
    }
```

```java
        // Processing rate metrics
        metrics.entrySet().stream()
            .filter(entry -> entry.getKey().name().equals("process-rate"))
            .forEach(entry -> {
                String threadId = entry.getKey().tags().get("thread-id");
                double rate = (Double) entry.getValue().metricValue();
                
                Gauge.builder("kafka.streams.process.rate")
                    .description("Processing rate per thread")
                    .tag("thread", threadId)
                    .register(meterRegistry)
                    .set(rate);
            });
        
        // Record processing latency
        metrics.entrySet().stream()
            .filter(entry -> entry.getKey().name().equals("process-latency-avg"))
            .forEach(entry -> {
                String threadId = entry.getKey().tags().get("thread-id");
                double latency = (Double) entry.getValue().metricValue();
                
                Gauge.builder("kafka.streams.process.latency")
                    .description("Average processing latency per thread")
                    .tag("thread", threadId)
                    .register(meterRegistry)
                    .set(latency);
            });
    }
    
    private void collectLagMetrics() {
        // Consumer lag monitoring
        try {
            kafkaStreams.allLocalStorePartitionLags().forEach((storeName, lags) -> {
                lags.forEach(lag -> {
                    Gauge.builder("kafka.streams.consumer.lag")
                        .description("Consumer lag for store partitions")
                        .tag("store", storeName)
                        .tag("partition", String.valueOf(lag.currentOffsetPosition()))
                        .register(meterRegistry)
                        .set(lag.lag());
                });
            });
        } catch (Exception e) {
            log.warn("Failed to collect lag metrics", e);
        }
    }
    
    private void collectStateStoreMetrics() {
        // State store size and performance metrics
        kafkaStreams.allLocalStorePartitionLags().keySet().forEach(storeName -> {

                    .set(approximateSize);
                
            } catch (Exception e) {
```

```java
@Component
public class DataLossDetector {
    
    private final KafkaTemplate<String, Object> debugKafkaTemplate;

    @KafkaListener(topics = "input-topic")

        MessageTracker.trackInput(record.key(), record.offset(), record.partition());
        
        // Add debugging headers
        debugKafkaTemplate.send("debug-input-tracking", record.key(), 
            DebugMessage.builder()
                .messageId(record.key())
                .partition(record.partition())
                .offset(record.offset())
                .timestamp(System.currentTimeMillis())
                .stage("INPUT")
                .build());
    }
    
    @KafkaListener(topics = "output-topic")  
    public void trackOutputMessages(ConsumerRecord<String, Object> record) {
        // Track output message
        boolean wasTracked = MessageTracker.trackOutput(record.key());
        
        if (!wasTracked) {
            // Potential data integrity issue
            log.error("Output message {} was not tracked in input", record.key());
            dataLossCounter.increment();
        }
        
        // Debug output tracking
        debugKafkaTemplate.send("debug-output-tracking", record.key(),
            DebugMessage.builder()
                .messageId(record.key())
                .partition(record.partition()) 
                .offset(record.offset())
                .timestamp(System.currentTimeMillis())
                .stage("OUTPUT")
                .build());
    }
    
    @Scheduled(fixedDelay = 60000) // Every minute
    public void detectPotentialDataLoss() {
        List<String> potentialLostMessages = MessageTracker.findPotentialLostMessages(
            Duration.ofMinutes(5)
        );
        
        if (!potentialLostMessages.isEmpty()) {
            log.error("Potential data loss detected for messages: {}", potentialLostMessages);
            
            potentialLostMessages.forEach(messageId -> {
                dataLossCounter.increment();
                
                // Send alert
                alertService.sendAlert(Alert.builder()
                    .severity(AlertSeverity.HIGH)
                    .title("Potential Data Loss Detected")
                    .message(String.format("Message %s appears to be lost", messageId))
                    .timestamp(Instant.now())
                    .build());
            });
        }
    }
}

@Component
public class MessageTracker {
    private static final Cache<String, MessageTrace> messageTraces = 
        Caffeine.newBuilder()
            .maximumSize(100000)
            .expireAfterWrite(Duration.ofMinutes(10))
            .build();
    
    public static void trackInput(String messageId, long offset, int partition) {
        messageTraces.put(messageId, MessageTrace.builder()
            .messageId(messageId)
            .inputOffset(offset)
            .inputPartition(partition)
            .inputTimestamp(System.currentTimeMillis())
            .build());
    }
    
    public static boolean trackOutput(String messageId) {
        MessageTrace trace = messageTraces.getIfPresent(messageId);
        if (trace != null) {
            trace.setOutputTimestamp(System.currentTimeMillis());
            trace.setProcessed(true);
            return true;
        }
        return false;
    }
    
    public static List<String> findPotentialLostMessages(Duration timeWindow) {
        long cutoffTime = System.currentTimeMillis() - timeWindow.toMillis();
        
        return messageTraces.asMap().values().stream()
            .filter(trace -> trace.getInputTimestamp() < cutoffTime)
            .filter(trace -> !trace.isProcessed())
            .map(MessageTrace::getMessageId)
            .collect(Collectors.toList());
    }
}
```
