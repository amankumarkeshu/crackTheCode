#### Core Recommendation Pipeline

```python
from typing import List, Dict, Set, Tuple
from dataclasses import dataclass
from enum import Enum
import time

class RecommendationSignal(Enum):
    MUTUAL_CONNECTIONS = "mutual_connections"
    SAME_WORKPLACE = "same_workplace"  
    SAME_EDUCATION = "same_education"
    SAME_INDUSTRY = "same_industry"
    PROFILE_VIEWS = "profile_views"
    CONTACT_IMPORT = "contact_import"
    GEOGRAPHIC_PROXIMITY = "geographic"

@dataclass
class Member:
    member_id: str
    current_company: str
    industry: str
    location: str
    education_institutions: List[str]
    connection_count: int
    privacy_settings: Dict[str, bool]

@dataclass
class RecommendationCandidate:
    member_id: str
    signals: Dict[RecommendationSignal, float]
    overall_score: float
    explanation: str

class PYMKRecommendationEngine:
    def __init__(self):
        self.signal_weights = {
            RecommendationSignal.MUTUAL_CONNECTIONS: 0.35,
            RecommendationSignal.SAME_WORKPLACE: 0.25,
            RecommendationSignal.SAME_EDUCATION: 0.15,
            RecommendationSignal.PROFILE_VIEWS: 0.10,
            RecommendationSignal.SAME_INDUSTRY: 0.08,
            RecommendationSignal.CONTACT_IMPORT: 0.05,
            RecommendationSignal.GEOGRAPHIC_PROXIMITY: 0.02
        }
        self.graph_service = SocialGraphService()
        self.privacy_service = PrivacyService()
        
    def get_recommendations(self, member_id: str, limit: int = 20) -> List[RecommendationCandidate]:
        """
        Generate PYMK recommendations for a member
        """
        member = self._get_member_profile(member_id)
        if not member:
            return []
        
        # Get candidate pool from multiple signal sources
        candidates = set()
        
        # 1. Mutual connections (strongest signal)
        mutual_candidates = self._get_mutual_connection_candidates(member_id)
        candidates.update(mutual_candidates)
        
        # 2. Workplace connections
        workplace_candidates = self._get_workplace_candidates(member)
        candidates.update(workplace_candidates)
        
        # 3. Education connections  
        education_candidates = self._get_education_candidates(member)
        candidates.update(education_candidates)
        
        # 4. Industry connections
        industry_candidates = self._get_industry_candidates(member)
        candidates.update(industry_candidates)
        
        # Filter out existing connections and privacy restrictions
        filtered_candidates = self._filter_candidates(member_id, candidates)
        
        # Score each candidate
        scored_candidates = []
        for candidate_id in filtered_candidates:
            candidate = self._score_candidate(member, candidate_id)
            if candidate and candidate.overall_score > 0.3:  # Minimum threshold
                scored_candidates.append(candidate)
        
        # Sort by score and return top recommendations
        scored_candidates.sort(key=lambda x: x.overall_score, reverse=True)
        return scored_candidates[:limit]
    
    def _get_mutual_connection_candidates(self, member_id: str) -> Set[str]:
        """Find friends-of-friends candidates"""
        direct_connections = self.graph_service.get_connections(member_id)
        candidates = set()
        
        for connection_id in direct_connections:
            # Get connections of connections (2-hop neighbors)
            second_degree = self.graph_service.get_connections(connection_id)
            candidates.update(second_degree)
        
        # Remove direct connections and self
        candidates.discard(member_id)
        candidates -= set(direct_connections)
        
        return candidates
    
    def _score_candidate(self, member: Member, candidate_id: str) -> RecommendationCandidate:
        """Calculate recommendation score for a candidate"""
        candidate_profile = self._get_member_profile(candidate_id)
        if not candidate_profile:
            return None
        
        signals = {}
        
        # Mutual connections signal
        mutual_count = len(self._get_mutual_connections(member.member_id, candidate_id))
        signals[RecommendationSignal.MUTUAL_CONNECTIONS] = min(1.0, mutual_count / 10.0)
        
        # Workplace signal
        workplace_match = member.current_company == candidate_profile.current_company
        signals[RecommendationSignal.SAME_WORKPLACE] = 1.0 if workplace_match else 0.0
        
        # Education signal
        education_overlap = set(member.education_institutions) & set(candidate_profile.education_institutions)
        signals[RecommendationSignal.SAME_EDUCATION] = 1.0 if education_overlap else 0.0
        
        # Industry signal
        industry_match = member.industry == candidate_profile.industry
        signals[RecommendationSignal.SAME_INDUSTRY] = 1.0 if industry_match else 0.0
        
        # Profile views signal (would come from analytics service)
        view_score = self._get_profile_view_signal(member.member_id, candidate_id)
        signals[RecommendationSignal.PROFILE_VIEWS] = view_score
        
        # Calculate weighted overall score
        overall_score = 0.0
        for signal, value in signals.items():
            weight = self.signal_weights.get(signal, 0.0)
            overall_score += weight * value
        
        # Generate explanation for transparency
        explanation = self._generate_explanation(signals, candidate_profile)
        
        return RecommendationCandidate(
            member_id=candidate_id,
            signals=signals,
            overall_score=overall_score,
            explanation=explanation
        )
    
    def _filter_candidates(self, member_id: str, candidates: Set[str]) -> Set[str]:
        """Filter candidates based on privacy settings and existing connections"""
        filtered = set()
        
        existing_connections = set(self.graph_service.get_connections(member_id))
        
        for candidate_id in candidates:
            # Skip if already connected
            if candidate_id in existing_connections:
                continue
            
            # Check privacy settings
            if not self.privacy_service.can_be_recommended(member_id, candidate_id):
                continue
            
            filtered.add(candidate_id)
        
        return filtered
    
    def _generate_explanation(self, signals: Dict[RecommendationSignal, float], candidate: Member) -> str:
        """Generate human-readable explanation for recommendation"""
        explanations = []
        
        if signals.get(RecommendationSignal.MUTUAL_CONNECTIONS, 0) > 0.3:
            explanations.append("You have mutual connections")
        
        if signals.get(RecommendationSignal.SAME_WORKPLACE, 0) > 0:
            explanations.append(f"You both work at {candidate.current_company}")
        
        if signals.get(RecommendationSignal.SAME_EDUCATION, 0) > 0:
            explanations.append("You went to the same school")
        
        if signals.get(RecommendationSignal.SAME_INDUSTRY, 0) > 0:
            explanations.append(f"You both work in {candidate.industry}")
        
        return " • ".join(explanations) if explanations else "Based on your professional network"
```

```python
class PrivacyService:
    def can_be_recommended(self, recommender_id: str, candidate_id: str) -> bool:
        """Check if candidate can be recommended to recommender"""
        
        candidate_settings = self._get_privacy_settings(candidate_id)
        
        # Check if candidate allows discovery
        if not candidate_settings.get('allow_discovery', True):
            return False
        
        # Check if candidate blocks recommendations from this industry/company
        recommender_profile = self._get_member_profile(recommender_id)
        blocked_companies = candidate_settings.get('blocked_companies', [])
        if recommender_profile.current_company in blocked_companies:
            return False
        
        # Check mutual block relationships
        if self._are_members_blocked(recommender_id, candidate_id):
            return False
        
        return True
```

#### Problem 1: Connection Strength Modeling (45 minutes)

**Question**: Design an algorithm to calculate relationship strength between LinkedIn members for feed ranking purposes.

**My Approach**: Multi-signal relationship strength model

```python
from typing import Dict, List, Optional
from dataclasses import dataclass
from datetime import datetime, timedelta
import math

@dataclass
class InteractionEvent:
    event_type: str  # message, like, comment, profile_view, etc.
    timestamp: datetime
    weight: float

class ConnectionStrengthCalculator:
    def __init__(self):
        self.interaction_weights = {
            'message': 10.0,
            'comment': 5.0, 
            'like': 2.0,
            'share': 7.0,
            'profile_view': 1.0,
            'endorsement': 8.0,
            'recommendation': 15.0
        }
        self.decay_rate = 0.1  # Decay rate per day
    
    def calculate_strength(self, member_a: str, member_b: str) -> float:
        """
        Calculate relationship strength between two members
        Range: 0.0 (no relationship) to 1.0 (very strong relationship)
        """
        
        # Get interaction history
        interactions = self._get_interaction_history(member_a, member_b)
        
        if not interactions:
            return 0.0
        
        # Calculate interaction score with time decay
        interaction_score = self._calculate_interaction_score(interactions)
        
        # Get structural signals
        structural_score = self._calculate_structural_signals(member_a, member_b)
        
        # Get professional similarity
        professional_score = self._calculate_professional_similarity(member_a, member_b)
        
        # Combine signals
        raw_score = (
            interaction_score * 0.6 +      # Interactions are most important
            structural_score * 0.25 +      # Network structure matters
            professional_score * 0.15      # Professional similarity
        )
        
        # Apply sigmoid normalization to get 0-1 range
        normalized_score = 1.0 / (1.0 + math.exp(-raw_score))
        
        return normalized_score
    
    def _calculate_interaction_score(self, interactions: List[InteractionEvent]) -> float:
        """Calculate score based on interaction history with time decay"""
        current_time = datetime.now()
        total_score = 0.0
        
        for interaction in interactions:
            # Base score from interaction type
            base_score = self.interaction_weights.get(interaction.event_type, 0.0)
            
            # Apply time decay (interactions decay over time)
            days_ago = (current_time - interaction.timestamp).days
            decay_factor = math.exp(-self.decay_rate * days_ago)
            
            # Apply additional event-specific weight
            weighted_score = base_score * interaction.weight * decay_factor
            
            total_score += weighted_score
        
        return total_score
    
    def _calculate_structural_signals(self, member_a: str, member_b: str) -> float:
        """Calculate strength based on network structure"""
        score = 0.0
        
        # Mutual connections (Jaccard similarity)
        connections_a = set(self._get_connections(member_a))
        connections_b = set(self._get_connections(member_b))
        
        if connections_a or connections_b:
            jaccard_similarity = len(connections_a & connections_b) / len(connections_a | connections_b)
            score += jaccard_similarity * 10.0
        
        # Path length (shorter paths indicate stronger relationships)
        path_length = self._get_shortest_path_length(member_a, member_b)
        if path_length == 1:  # Direct connection
            score += 5.0
        elif path_length == 2:  # 2 hops
            score += 2.0
        elif path_length == 3:  # 3 hops  
            score += 0.5
        
        return score
    
    def _calculate_professional_similarity(self, member_a: str, member_b: str) -> float:
        """Calculate professional similarity score"""
        profile_a = self._get_member_profile(member_a)
        profile_b = self._get_member_profile(member_b)
        
        score = 0.0
        
        # Same company
        if profile_a.current_company == profile_b.current_company:
            score += 3.0
        
        # Same industry
        if profile_a.industry == profile_b.industry:
            score += 1.5
        
        # Similar job titles (would use NLP similarity in production)
        title_similarity = self._calculate_title_similarity(profile_a.job_title, profile_b.job_title)
        score += title_similarity * 2.0
        
        # Same education institutions
        education_overlap = set(profile_a.education) & set(profile_b.education)
        score += len(education_overlap) * 1.0
        
        return score

# Usage for feed ranking
class FeedRankingService:
    def __init__(self):
        self.strength_calculator = ConnectionStrengthCalculator()
    
    def rank_feed_items(self, member_id: str, feed_items: List[FeedItem]) -> List[FeedItem]:
        """Rank feed items based on connection strength and other signals"""
        
        scored_items = []
        
        for item in feed_items:
            # Get connection strength to content author
            strength = self.strength_calculator.calculate_strength(member_id, item.author_id)
            
            # Calculate overall feed ranking score
            ranking_score = self._calculate_feed_score(item, strength)
            
            scored_items.append((item, ranking_score))
        
        # Sort by score and return ranked items
        scored_items.sort(key=lambda x: x[1], reverse=True)
        return [item for item, score in scored_items]
```

```python
class ViralContentDetector:
    def __init__(self):
        self.engagement_thresholds = {
            'views': {'1h': 1000, '6h': 10000, '24h': 100000},
            'likes': {'1h': 50, '6h': 500, '24h': 5000}, 
            'comments': {'1h': 10, '6h': 100, '24h': 1000},
            'shares': {'1h': 20, '6h': 200, '24h': 2000}
        }
    
    def detect_viral_potential(self, post_id: str) -> Dict[str, float]:
        """
        Analyze post for viral potential
        Returns viral score and contributing factors
        """
        post = self._get_post_details(post_id)
        if not post:
            return {'viral_score': 0.0, 'factors': {}}
        
        # Engagement velocity analysis
        velocity_score = self._calculate_engagement_velocity(post)
        
        # Content quality signals
        content_score = self._analyze_content_quality(post)
        
        # Network reach potential
        network_score = self._calculate_network_reach(post)
        
        # Professional relevance
        relevance_score = self._calculate_professional_relevance(post)
        
        # Authority of author
        authority_score = self._calculate_author_authority(post.author_id)
        
        # Combine signals
        viral_score = (
            velocity_score * 0.35 +     # Engagement velocity most important
            content_score * 0.25 +      # Content quality
            network_score * 0.20 +      # Network reach
            relevance_score * 0.15 +    # Professional relevance
            authority_score * 0.05      # Author authority
        )
        
        return {
            'viral_score': min(1.0, viral_score),
            'factors': {
                'velocity': velocity_score,
                'content_quality': content_score,
                'network_reach': network_score,
                'professional_relevance': relevance_score,
                'author_authority': authority_score
            }
        }
    
    def _calculate_engagement_velocity(self, post: Post) -> float:
        """Calculate how fast engagement is growing"""
        current_time = datetime.now()
        post_age_hours = (current_time - post.created_at).total_seconds() / 3600
        
        if post_age_hours < 1:
            return 0.0  # Too early to judge
        
        # Get engagement metrics
        metrics = self._get_engagement_metrics(post.post_id)
        
        velocity_score = 0.0
        
        # Check against thresholds for different time windows
        for time_window in ['1h', '6h', '24h']:
            window_hours = int(time_window[:-1])
            
            if post_age_hours >= window_hours:
                for metric_type, threshold in self.engagement_thresholds[time_window].items():
                    actual_value = metrics.get(metric_type, 0)
                    if actual_value >= threshold:
                        velocity_score += 1.0
        
        return velocity_score / 12.0  # Normalize to 0-1 range
    
    def _calculate_professional_relevance(self, post: Post) -> float:
        """Score content for professional relevance"""
        
        # Content analysis (simplified - would use NLP in production)
        professional_keywords = [
            'career', 'leadership', 'business', 'strategy', 'innovation',
            'management', 'entrepreneurship', 'skills', 'industry', 'market'
        ]
        
        content_text = (post.title + ' ' + post.content).lower()
        keyword_matches = sum(1 for keyword in professional_keywords if keyword in content_text)
        keyword_score = min(1.0, keyword_matches / 5.0)
        
        # Post type relevance
        type_scores = {
            'career_advice': 0.9,
            'industry_insights': 0.8, 
            'company_updates': 0.7,
            'personal_achievement': 0.6,
            'thought_leadership': 0.9,
            'job_posting': 0.5,
            'personal_story': 0.4
        }
        
        type_score = type_scores.get(post.post_type, 0.3)
        
        return (keyword_score * 0.6 + type_score * 0.4)
```

```python
class CandidateSelectionModel:
    """
    Fast retrieval model to select relevant candidates from millions of posts
    Uses lightweight features and approximate algorithms for speed
    """
    
    def __init__(self):
        self.connection_graph = ConnectionGraph()
        self.content_index = ContentIndex()
        
    def select_candidates(self, member_id: str, candidate_limit: int) -> List[FeedCandidate]:
        """Select candidate posts using multiple retrieval strategies"""
        
        candidates = set()
        
        # Strategy 1: Network-based retrieval (60% of candidates)
        network_candidates = self._get_network_content(member_id, int(candidate_limit * 0.6))
        candidates.update(network_candidates)
        
        # Strategy 2: Interest-based retrieval (30% of candidates)  
        interest_candidates = self._get_interest_based_content(member_id, int(candidate_limit * 0.3))
        candidates.update(interest_candidates)
        
        # Strategy 3: Trending content (10% of candidates)
        trending_candidates = self._get_trending_content(member_id, int(candidate_limit * 0.1))
        candidates.update(trending_candidates)
        
        return list(candidates)[:candidate_limit]
    
    def _get_network_content(self, member_id: str, limit: int) -> List[FeedCandidate]:
        """Get content from member's professional network"""
        
        # Get 1st and 2nd degree connections
        connections_1st = self.connection_graph.get_direct_connections(member_id)
        connections_2nd = self.connection_graph.get_second_degree_connections(member_id, limit=500)
        
        all_connections = connections_1st + connections_2nd[:100]  # Limit 2nd degree
        
        # Get recent content from these connections
        candidates = []
        for connection_id in all_connections:
            recent_posts = self.content_index.get_recent_posts(
                author_id=connection_id,
                since=datetime.now() - timedelta(days=7),
                limit=5
            )
            candidates.extend(recent_posts)
        
        # Sort by recency and engagement
        candidates.sort(key=lambda x: (x.created_time, x.features.get('engagement_score', 0)), reverse=True)
        
        return candidates[:limit]
    
    def _get_interest_based_content(self, member_id: str, limit: int) -> List[FeedCandidate]:
        """Get content based on member's professional interests"""
        
        member_profile = self._get_member_profile(member_id)
        
        # Get content matching member's industry, skills, interests
        industry_content = self.content_index.get_content_by_industry(
            industry=member_profile.industry,
            limit=limit//2
        )
        
        skill_content = self.content_index.get_content_by_skills(
            skills=member_profile.skills,
            limit=limit//2
        )
        
        return industry_content + skill_content

class DeepRankingModel:
    """
    Precise ranking model using deep learning for final ranking
    Uses rich features and complex interactions
    """
    
    def __init__(self):
        self.model = self._load_trained_model()
        self.feature_extractor = FeatureExtractor()
        
    def rank_candidates(self, member_id: str, candidates: List[FeedCandidate], limit: int) -> List[FeedCandidate]:
        """Rank candidates using deep learning model"""
        
        # Extract rich features for each candidate
        feature_vectors = []
        for candidate in candidates:
            features = self.feature_extractor.extract_features(member_id, candidate)
            feature_vectors.append(features)
        
        # Batch prediction for efficiency
        if feature_vectors:
            scores = self.model.predict(np.array(feature_vectors))
            
            # Add scores to candidates and sort
            scored_candidates = list(zip(candidates, scores))
            scored_candidates.sort(key=lambda x: x[1], reverse=True)
            
            return [candidate for candidate, score in scored_candidates[:limit]]
        
        return candidates[:limit]

class FeatureExtractor:

    def extract_features(self, member_id: str, candidate: FeedCandidate) -> np.ndarray:

        member_features = self._extract_member_features(member_id)

        content_features = self._extract_content_features(candidate)
        
        # Interaction features (member-content specific)
        interaction_features = self._extract_interaction_features(member_id, candidate)
        
        # Contextual features
        contextual_features = self._extract_contextual_features(member_id, candidate)
        
        # Combine all features
        all_features = np.concatenate([
            member_features,
            content_features, 
            interaction_features,
            contextual_features

    def _extract_member_features(self, member_id: str) -> np.ndarray:

            profile.seniority_level,  # 0-5 scale
            profile.connection_count / 500.0,  # Normalized
            profile.follower_count / 1000.0,  # Normalized
            len(profile.skills) / 20.0,  # Normalized
            profile.activity_score,  # 0-1 scale
        ]

        industry_encoding = self._one_hot_encode(profile.industry, self.industry_categories)
        company_size_encoding = self._one_hot_encode(profile.company_size, self.company_size_buckets)

    def _extract_content_features(self, candidate: FeedCandidate) -> np.ndarray:
        """Extract content-specific features"""
        
        features = [
            candidate.features.get('engagement_rate', 0.0),
            candidate.features.get('professional_score', 0.0),
            candidate.features.get('viral_score', 0.0),
            candidate.features.get('quality_score', 0.0),
            self._get_content_age_hours(candidate) / 168.0,  # Normalize to week
        ]
        
        # Content type encoding
        type_encoding = self._one_hot_encode(candidate.content_type, self.content_types)
        
        return np.concatenate([features, type_encoding])
    
    def _extract_interaction_features(self, member_id: str, candidate: FeedCandidate) -> np.ndarray:
        """Extract member-content interaction features"""
        
        # Connection strength to author
        connection_strength = self._get_connection_strength(member_id, candidate.author_id)

        similar_engagement = self._get_similar_content_engagement(member_id, candidate)
        
        # Author interaction history
```

```python
class OptimizedFeatureExtractor:
    def __init__(self):
        self.feature_cache = RedisCache('features', ttl=900)  # 15 min TTL
        self.batch_size = 50
        
    async def extract_features_batch(self, member_id: str, candidates: List[FeedCandidate]) -> List[np.ndarray]:
        """Batch feature extraction with caching"""
        
        cached_features = []
        uncached_candidates = []
        
        # Check cache for existing features
        cache_keys = [f"features:{member_id}:{c.content_id}" for c in candidates]
        cached_results = await self.feature_cache.mget(cache_keys)
        
        for i, cached_result in enumerate(cached_results):
            if cached_result:
                cached_features.append((i, cached_result))
            else:
                uncached_candidates.append((i, candidates[i]))
        
        # Batch compute uncached features
        if uncached_candidates:
            batch_features = await self._compute_features_batch(member_id, uncached_candidates)
            
            # Cache computed features
            cache_operations = []
            for (idx, candidate), features in zip(uncached_candidates, batch_features):
                cache_key = f"features:{member_id}:{candidate.content_id}"
                cache_operations.append(self.feature_cache.set(cache_key, features))
            
            await asyncio.gather(*cache_operations)
        
        # Combine cached and computed features in correct order
        all_features = [None] * len(candidates)
        
        for idx, features in cached_features:
            all_features[idx] = features
            
        for (idx, _), features in zip(uncached_candidates, batch_features if uncached_candidates else []):
            all_features[idx] = features
        
        return all_features
    
    async def _compute_features_batch(self, member_id: str, candidate_batch: List[Tuple[int, FeedCandidate]]) -> List[np.ndarray]:
        """Compute features in batches for efficiency"""
```
