```python
from typing import List, Dict, Optional, Set
from dataclasses import dataclass
from datetime import datetime, timedelta
import pytz
from enum import Enum

class MeetingStatus(Enum):
    SCHEDULED = "scheduled"
    CONFIRMED = "confirmed"  
    CANCELLED = "cancelled"
    RESCHEDULED = "rescheduled"

class ResourceType(Enum):
    MEETING_ROOM = "room"
    PROJECTOR = "projector"
    VIDEO_CONFERENCE = "video_conf"
    WHITEBOARD = "whiteboard"

@dataclass
class Resource:
    resource_id: str
    resource_type: ResourceType
    location: str
    capacity: int
    available_hours: tuple  # (start_hour, end_hour) in local time

@dataclass
class Participant:
    user_id: str
    email: str
    name: str
    timezone: str
    working_hours: tuple  # (start_hour, end_hour) in their timezone
    
@dataclass
class Meeting:
    meeting_id: str
    title: str
    organizer_id: str
    participants: List[Participant]
    start_time: datetime
    end_time: datetime
    timezone: str
    required_resources: List[ResourceType]
    is_recurring: bool = False
    recurrence_pattern: Optional[str] = None  # "weekly", "daily", etc.
    status: MeetingStatus = MeetingStatus.SCHEDULED

class MicrosoftTeamsMeetingScheduler:
    def __init__(self):
        self.meetings = {}  # meeting_id -> Meeting
        self.resources = {}  # resource_id -> Resource  
        self.user_calendars = {}  # user_id -> List[Meeting]
        self.resource_bookings = {}  # resource_id -> List[(start_time, end_time)]
        self.timezone_cache = {}  # Cache for timezone conversions
        
    def schedule_meeting(self, meeting: Meeting) -> Dict[str, any]:
        """
        Schedule a meeting with conflict resolution and resource allocation
        """
        try:
            # Validate participants' availability  
            availability_conflicts = self._check_participant_availability(meeting)
            if availability_conflicts:
                return self._suggest_alternative_times(meeting, availability_conflicts)
            
            # Allocate required resources
            resource_allocation = self._allocate_resources(meeting)
            if not resource_allocation['success']:
                return {
                    'success': False,
                    'message': 'Required resources not available',
                    'alternatives': resource_allocation['alternatives']
                }
            
            # Handle recurring meetings
            if meeting.is_recurring:
                return self._schedule_recurring_meeting(meeting, resource_allocation['resources'])
            
            # Schedule single meeting
            meeting.meeting_id = self._generate_meeting_id()
            self.meetings[meeting.meeting_id] = meeting
            
            # Update participant calendars
            for participant in meeting.participants:
                if participant.user_id not in self.user_calendars:
                    self.user_calendars[participant.user_id] = []
                self.user_calendars[participant.user_id].append(meeting)
            
            # Reserve resources
            for resource in resource_allocation['resources']:
                self._reserve_resource(resource, meeting.start_time, meeting.end_time)
            
            meeting.status = MeetingStatus.CONFIRMED
            
            return {
                'success': True,
                'meeting_id': meeting.meeting_id,
                'meeting_link': self._generate_teams_link(meeting.meeting_id),
                'allocated_resources': resource_allocation['resources'],
                'calendar_invites': self._generate_calendar_invites(meeting)
            }
            
        except Exception as e:
            return {
                'success': False,
                'message': f'Scheduling error: {str(e)}'
            }
    
    def _check_participant_availability(self, meeting: Meeting) -> List[Dict]:
        """
        Check all participants' availability across time zones
        """
        conflicts = []
        
        for participant in meeting.participants:
            # Convert meeting time to participant's timezone
            participant_tz = pytz.timezone(participant.timezone)
            meeting_start_local = meeting.start_time.astimezone(participant_tz)
            meeting_end_local = meeting.end_time.astimezone(participant_tz)
            
            # Check working hours
            if not self._is_within_working_hours(participant, meeting_start_local, meeting_end_local):
                conflicts.append({
                    'participant': participant.user_id,
                    'type': 'outside_working_hours',
                    'participant_timezone': participant.timezone,
                    'local_meeting_time': {
                        'start': meeting_start_local.isoformat(),
                        'end': meeting_end_local.isoformat()
                    }
                })
            
            # Check existing calendar conflicts
            existing_meetings = self.user_calendars.get(participant.user_id, [])
            for existing_meeting in existing_meetings:
                if self._meetings_overlap(meeting, existing_meeting):
                    conflicts.append({
                        'participant': participant.user_id,
                        'type': 'calendar_conflict',
                        'conflicting_meeting': existing_meeting.meeting_id,
                        'conflict_details': {
                            'existing_start': existing_meeting.start_time.isoformat(),
                            'existing_end': existing_meeting.end_time.isoformat()
                        }
                    })
        
        return conflicts
    
    def _allocate_resources(self, meeting: Meeting) -> Dict:
        """
        Allocate required resources for the meeting
        """
        allocated_resources = []
        unavailable_resources = []
        
        for required_resource_type in meeting.required_resources:
            available_resource = self._find_available_resource(
                required_resource_type, 
                meeting.start_time, 
                meeting.end_time,
                len(meeting.participants)
            )
            
            if available_resource:
                allocated_resources.append(available_resource)
            else:
                unavailable_resources.append(required_resource_type)
                
                # Find alternative resources
                alternatives = self._find_alternative_resources(
                    required_resource_type,
                    meeting.start_time,
                    meeting.end_time
                )
                
        if unavailable_resources:
            return {
                'success': False,
                'unavailable': unavailable_resources,
                'alternatives': self._suggest_alternative_resources(meeting)
            }
        
        return {
            'success': True,
            'resources': allocated_resources
        }
    
    def _find_available_resource(self, resource_type: ResourceType, 
                               start_time: datetime, end_time: datetime,
                               participant_count: int) -> Optional[Resource]:
        """
        Find available resource of specified type for given time slot
        """
        for resource_id, resource in self.resources.items():
            if (resource.resource_type == resource_type and
                resource.capacity >= participant_count and
                self._is_resource_available(resource_id, start_time, end_time)):
                return resource
        
        return None
    
    def _is_resource_available(self, resource_id: str, 
                             start_time: datetime, end_time: datetime) -> bool:
        """
        Check if resource is available for the specified time period
        """
        if resource_id not in self.resource_bookings:
            return True
        
        bookings = self.resource_bookings[resource_id]
        
        for booking_start, booking_end in bookings:
            if (start_time < booking_end and end_time > booking_start):
                return False  # Overlap found
        
        return True
    
    def _suggest_alternative_times(self, meeting: Meeting, 
                                 conflicts: List[Dict]) -> Dict:
        """
        Suggest alternative meeting times when conflicts exist
        """
        # Find common free slots for all participants
        base_time = meeting.start_time
        duration = meeting.end_time - meeting.start_time
        
        alternative_slots = []
        
        # Check next 7 days for available slots
        for day_offset in range(7):
            check_date = base_time + timedelta(days=day_offset)
            
            # Check hourly slots during business hours
            for hour in range(9, 18):  # 9 AM to 6 PM
                potential_start = check_date.replace(
                    hour=hour, minute=0, second=0, microsecond=0
                )
                potential_end = potential_start + duration
                
                # Create temporary meeting for conflict checking
                temp_meeting = Meeting(
                    "", meeting.title, meeting.organizer_id,
                    meeting.participants, potential_start, potential_end,
                    meeting.timezone, meeting.required_resources
                )
                
                # Check if this slot works for everyone
                temp_conflicts = self._check_participant_availability(temp_meeting)
                resource_check = self._allocate_resources(temp_meeting)
                
                if not temp_conflicts and resource_check['success']:
                    alternative_slots.append({
                        'start_time': potential_start.isoformat(),
                        'end_time': potential_end.isoformat(),
                        'all_participants_available': True,
                        'resources_available': True
                    })
                
                if len(alternative_slots) >= 5:  # Limit to 5 suggestions
                    break
            
            if len(alternative_slots) >= 5:
                break
        
        return {
            'success': False,
            'message': 'Conflicts found with requested time',
            'conflicts': conflicts,
            'alternative_times': alternative_slots
        }
    
    def _schedule_recurring_meeting(self, meeting: Meeting, 
                                  resources: List[Resource]) -> Dict:
        """
        Schedule recurring meetings with pattern handling
        """
        scheduled_instances = []
        failed_instances = []
        
        # Generate recurring instances (simplified for interview)
        recurrence_count = 10  # Next 10 occurrences
        
        if meeting.recurrence_pattern == "weekly":
            delta = timedelta(weeks=1)
        elif meeting.recurrence_pattern == "daily":
            delta = timedelta(days=1)
        else:
            delta = timedelta(weeks=1)  # Default to weekly
        
        current_start = meeting.start_time
        
        for i in range(recurrence_count):
            instance_start = current_start + (delta * i)
            instance_end = meeting.end_time + (delta * i)
            
            # Create meeting instance
            instance = Meeting(
                f"{meeting.meeting_id}_instance_{i}",
                f"{meeting.title} (Instance {i+1})",
                meeting.organizer_id,
                meeting.participants,
                instance_start,
                instance_end,
                meeting.timezone,
                meeting.required_resources,
                is_recurring=False
            )
            
            # Try to schedule this instance
            result = self.schedule_meeting(instance)
            
            if result['success']:
                scheduled_instances.append({
                    'instance_id': instance.meeting_id,
                    'start_time': instance_start.isoformat(),
                    'meeting_link': result['meeting_link']
                })
            else:
                failed_instances.append({
                    'instance_number': i + 1,
                    'start_time': instance_start.isoformat(),
                    'failure_reason': result['message']
                })
        
        return {
            'success': True,
            'recurring_series_id': meeting.meeting_id,
            'scheduled_instances': scheduled_instances,
            'failed_instances': failed_instances,
            'total_scheduled': len(scheduled_instances)
        }
    
    def reschedule_meeting(self, meeting_id: str, new_start_time: datetime, 
                          new_end_time: datetime) -> Dict:
        """
        Reschedule existing meeting with conflict checking
        """
        if meeting_id not in self.meetings:
            return {'success': False, 'message': 'Meeting not found'}
        
        original_meeting = self.meetings[meeting_id]
        
        # Create updated meeting object
        updated_meeting = Meeting(
            meeting_id,
            original_meeting.title,
            original_meeting.organizer_id,
            original_meeting.participants,
            new_start_time,
            new_end_time,
            original_meeting.timezone,
            original_meeting.required_resources
        )
        
        # Check availability for new time
        conflicts = self._check_participant_availability(updated_meeting)
        if conflicts:
            return self._suggest_alternative_times(updated_meeting, conflicts)
        
        # Free original resources and allocate new ones
        self._free_meeting_resources(original_meeting)
        resource_allocation = self._allocate_resources(updated_meeting)
        
        if not resource_allocation['success']:
            # Re-allocate original resources if new allocation fails
            self._allocate_resources(original_meeting)
            return {
                'success': False,
                'message': 'New time slot resources unavailable',
                'alternatives': resource_allocation['alternatives']
            }
        
        # Update meeting
        self.meetings[meeting_id] = updated_meeting
        updated_meeting.status = MeetingStatus.RESCHEDULED
        
        return {
            'success': True,
            'meeting_id': meeting_id,
            'new_start_time': new_start_time.isoformat(),
            'new_end_time': new_end_time.isoformat(),
            'updated_meeting_link': self._generate_teams_link(meeting_id)
        }
    
    # Helper methods
    def _is_within_working_hours(self, participant: Participant, 
                               start_time: datetime, end_time: datetime) -> bool:
        """Check if meeting time falls within participant's working hours"""
        start_hour = start_time.hour
        end_hour = end_time.hour
        
        working_start, working_end = participant.working_hours
        
        return (start_hour >= working_start and end_hour <= working_end)
    
    def _meetings_overlap(self, meeting1: Meeting, meeting2: Meeting) -> bool:
        """Check if two meetings have time overlap"""
        return (meeting1.start_time < meeting2.end_time and 
                meeting1.end_time > meeting2.start_time)
    
    def _generate_meeting_id(self) -> str:
        """Generate unique meeting ID"""
        import uuid
        return f"teams_meeting_{uuid.uuid4().hex[:8]}"
    
    def _generate_teams_link(self, meeting_id: str) -> str:
        """Generate Microsoft Teams meeting link"""
        return f"https://teams.microsoft.com/l/meetup-join/{meeting_id}"
    
    def _generate_calendar_invites(self, meeting: Meeting) -> List[Dict]:
        """Generate calendar invite data for all participants"""
        invites = []
        
        for participant in meeting.participants:
            participant_tz = pytz.timezone(participant.timezone)
            local_start = meeting.start_time.astimezone(participant_tz)
            local_end = meeting.end_time.astimezone(participant_tz)
            
            invites.append({
                'participant_email': participant.email,
                'subject': meeting.title,
                'start_time_local': local_start.isoformat(),
                'end_time_local': local_end.isoformat(),
                'timezone': participant.timezone,
                'teams_link': self._generate_teams_link(meeting.meeting_id)
            })
        
        return invites
```

#### Core Sync Algorithm

```python
from typing import Dict, List, Optional, Set
from dataclasses import dataclass
from datetime import datetime
from enum import Enum
import hashlib

class FileOperation(Enum):
    CREATE = "create"
    UPDATE = "update"
    DELETE = "delete"
    MOVE = "move"
    RENAME = "rename"

class ConflictType(Enum):
    CONCURRENT_EDIT = "concurrent_edit"
    MOVE_DELETE_CONFLICT = "move_delete"
    RENAME_CONFLICT = "rename_conflict"

@dataclass
class FileVersion:
    version_id: str
    file_id: str
    user_id: str
    device_id: str
    content_hash: str
    timestamp: datetime
    operation: FileOperation
    file_size: int
    parent_version_id: Optional[str] = None

@dataclass
class SyncOperation:
    operation_id: str
    file_id: str
    device_id: str
    operation_type: FileOperation
    content_hash: str
    timestamp: datetime
    metadata: Dict
    
class OneDriveFileSyncSystem:
    def __init__(self):
        self.file_versions = {}  # file_id -> List[FileVersion]
        self.device_sync_state = {}  # (device_id, file_id) -> version_id
        self.pending_operations = {}  # device_id -> List[SyncOperation]
        self.conflict_queue = {}  # file_id -> List[ConflictResolution]
        self.user_permissions = {}  # (user_id, file_id) -> permissions
        
    def sync_file_changes(self, device_id: str, operations: List[SyncOperation]) -> Dict:
        """
        Process file sync operations from a device
        """
        sync_results = []
        conflicts_detected = []
        
        for operation in operations:
            try:
                # Check permissions
                if not self._has_permission(operation.device_id, operation.file_id, operation.operation_type):
                    sync_results.append({
                        'operation_id': operation.operation_id,
                        'status': 'permission_denied',
                        'message': 'Insufficient permissions'
                    })
                    continue
                
                # Detect conflicts
                conflict = self._detect_conflicts(operation)
                
                if conflict:
                    conflicts_detected.append(conflict)
                    # Queue for conflict resolution
                    self._queue_for_conflict_resolution(operation, conflict)
                    
                    sync_results.append({
                        'operation_id': operation.operation_id,
                        'status': 'conflict_detected',
                        'conflict_type': conflict.type.value,
                        'resolution_options': self._get_conflict_resolution_options(conflict)
                    })
                else:
                    # Apply operation
                    result = self._apply_operation(operation)
                    sync_results.append(result)
                    
                    # Propagate to other devices
                    self._propagate_to_devices(operation)
                
            except Exception as e:
                sync_results.append({
                    'operation_id': operation.operation_id,
                    'status': 'error',
                    'message': str(e)
                })
        
        return {
            'sync_results': sync_results,
            'conflicts': conflicts_detected,
            'next_sync_token': self._generate_sync_token(device_id)
        }
    
    def _detect_conflicts(self, operation: SyncOperation) -> Optional['ConflictResolution']:
        """
        Detect conflicts with concurrent operations
        """
        file_id = operation.file_id
        
        # Get current file version
        current_version = self._get_current_version(file_id)
        device_last_sync = self._get_device_last_sync(operation.device_id, file_id)
        
        # Check if file was modified by another device since last sync
        if current_version and device_last_sync:
            if current_version.version_id != device_last_sync:
                # Concurrent modification detected
                
                if operation.operation_type == FileOperation.UPDATE:
                    # Both devices modified the file
                    return ConflictResolution(
                        conflict_id=f"conflict_{operation.operation_id}",
                        type=ConflictType.CONCURRENT_EDIT,
                        file_id=file_id,
                        conflicting_operations=[operation],
                        current_version=current_version,
                        device_version=device_last_sync
                    )
                
                elif operation.operation_type == FileOperation.DELETE:
                    if current_version.operation == FileOperation.UPDATE:
                        # Device A deleted, Device B modified
                        return ConflictResolution(
                            conflict_id=f"conflict_{operation.operation_id}",
                            type=ConflictType.MOVE_DELETE_CONFLICT,
                            file_id=file_id,
                            conflicting_operations=[operation],
                            current_version=current_version
                        )
        
        return None
    
    def _apply_operation(self, operation: SyncOperation) -> Dict:
        """
        Apply sync operation and create new file version
        """
        file_id = operation.file_id
        
        # Create new version
        new_version = FileVersion(
            version_id=self._generate_version_id(),
            file_id=file_id,
            user_id=self._get_user_from_device(operation.device_id),
            device_id=operation.device_id,
            content_hash=operation.content_hash,
            timestamp=operation.timestamp,
            operation=operation.operation_type,
            file_size=operation.metadata.get('file_size', 0),
            parent_version_id=self._get_current_version_id(file_id)
        )
        
        # Store version
        if file_id not in self.file_versions:
            self.file_versions[file_id] = []
        self.file_versions[file_id].append(new_version)
        
        # Update device sync state
        self.device_sync_state[(operation.device_id, file_id)] = new_version.version_id
        
        # Store file content (if applicable)
        storage_result = None
        if operation.operation_type in [FileOperation.CREATE, FileOperation.UPDATE]:
            storage_result = self._store_file_content(file_id, new_version.version_id, operation)
        
        return {
            'operation_id': operation.operation_id,
            'status': 'success',
            'new_version_id': new_version.version_id,
            'storage_url': storage_result.get('url') if storage_result else None
        }
    
    def _propagate_to_devices(self, operation: SyncOperation):
        """
        Propagate successful operation to other user devices
        """
        user_id = self._get_user_from_device(operation.device_id)
        user_devices = self._get_user_devices(user_id)
        
        # Send sync notifications to other devices
        for device_id in user_devices:
            if device_id != operation.device_id:
                self._send_sync_notification(device_id, operation)
    
    def resolve_conflict(self, conflict_id: str, resolution_choice: str, 
                        user_id: str) -> Dict:
        """
        Resolve detected conflict based on user choice
        """
        conflict = self._get_conflict(conflict_id)
        if not conflict:
            return {'success': False, 'message': 'Conflict not found'}
        
        try:
            if resolution_choice == 'keep_both':
                # Create renamed version of conflicting file
                result = self._resolve_keep_both(conflict)
                
            elif resolution_choice == 'keep_local':
                # Keep the device's version, discard server version
                result = self._resolve_keep_local(conflict)
                
            elif resolution_choice == 'keep_server':
                # Keep server version, discard device version
                result = self._resolve_keep_server(conflict)
                
            elif resolution_choice == 'manual_merge':
                # Create merge version (requires user input)
                result = self._resolve_manual_merge(conflict, user_id)
                
            else:
                return {'success': False, 'message': 'Invalid resolution choice'}
            
            # Remove from conflict queue
            self._remove_conflict(conflict_id)
            
            # Notify all devices of resolution
            self._notify_conflict_resolution(conflict, result)
            
            return {
                'success': True,
                'resolution': resolution_choice,
                'new_version_id': result.get('version_id'),
                'message': 'Conflict resolved successfully'
            }
            
        except Exception as e:
            return {'success': False, 'message': f'Resolution failed: {str(e)}'}
    
    def _resolve_keep_both(self, conflict: 'ConflictResolution') -> Dict:
        """
        Keep both versions by renaming one
        """
        file_id = conflict.file_id
        original_metadata = self._get_file_metadata(file_id)
        
        # Rename conflicting file
        conflicted_name = f"{original_metadata['name']} (Conflicted copy from {conflict.device_version})"
        
        # Create new file entry for conflicted version
        conflicted_file_id = self._generate_file_id()
        
        # Store both versions
        self._create_conflicted_copy(conflicted_file_id, conflicted_name, conflict)
        
        return {
            'version_id': conflict.current_version.version_id,
            'conflicted_file_id': conflicted_file_id,
            'action': 'created_conflicted_copy'
        }
    
    def get_file_history(self, file_id: str, user_id: str) -> List[Dict]:
        """
        Get version history for a file
        """
        if not self._has_permission(user_id, file_id, 'read'):
            return []
        
        versions = self.file_versions.get(file_id, [])
        
        history = []
        for version in sorted(versions, key=lambda v: v.timestamp, reverse=True):
            history.append({
                'version_id': version.version_id,
                'timestamp': version.timestamp.isoformat(),
                'operation': version.operation.value,
                'user_id': version.user_id,
                'device_id': version.device_id,
                'file_size': version.file_size,
                'download_url': self._get_version_download_url(file_id, version.version_id)
            })
        
        return history
    
    def restore_file_version(self, file_id: str, version_id: str, user_id: str) -> Dict:
        """
        Restore file to a previous version
        """
        if not self._has_permission(user_id, file_id, 'write'):
            return {'success': False, 'message': 'Permission denied'}
        
        target_version = self._get_version(file_id, version_id)
        if not target_version:
            return {'success': False, 'message': 'Version not found'}
        
        # Create restore operation
        restore_operation = SyncOperation(
            operation_id=self._generate_operation_id(),
            file_id=file_id,
            device_id='system',  # System-initiated restore
            operation_type=FileOperation.UPDATE,
            content_hash=target_version.content_hash,
            timestamp=datetime.utcnow(),
            metadata={'restored_from': version_id}
        )
        
        # Apply restore
        result = self._apply_operation(restore_operation)
        
        # Propagate to all devices
        self._propagate_to_devices(restore_operation)
        
        return {
            'success': True,
            'new_version_id': result['new_version_id'],
            'restored_from': version_id
        }
    
    # Helper methods
    def _generate_version_id(self) -> str:
        import uuid
        return f"ver_{uuid.uuid4().hex[:12]}"
    
    def _generate_operation_id(self) -> str:
        import uuid
        return f"op_{uuid.uuid4().hex[:8]}"
    
    def _get_current_version(self, file_id: str) -> Optional[FileVersion]:
        versions = self.file_versions.get(file_id, [])
        return max(versions, key=lambda v: v.timestamp) if versions else None
    
    def _store_file_content(self, file_id: str, version_id: str, 
                          operation: SyncOperation) -> Dict:
        # Store file content in Azure Blob Storage
        blob_path = f"files/{file_id}/versions/{version_id}"
        
        return {
            'url': f"https://onedrive.blob.core.windows.net/{blob_path}",
            'storage_path': blob_path
        }
```
