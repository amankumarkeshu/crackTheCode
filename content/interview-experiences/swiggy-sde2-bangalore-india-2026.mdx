    def _calculate_distance(self, loc1: Location, loc2: Location) -> float:
        """Calculate distance between two locations using Haversine formula"""
        
        lat1, lon1 = math.radians(loc1.latitude), math.radians(loc1.longitude)
        lat2, lon2 = math.radians(loc2.latitude), math.radians(loc2.longitude)
        
        dlat = lat2 - lat1
        dlon = lon2 - lon1
        
        a = math.sin(dlat/2)**2 + math.cos(lat1) * math.cos(lat2) * math.sin(dlon/2)**2
        c = 2 * math.asin(math.sqrt(a))
        
        # Radius of earth in kilometers
        r = 6371
        
        return c * r

class SwiggyRealTimeTrackingSystem:

        self.location_processor = LocationProcessor()
        self.eta_calculator = ETACalculator()
        self.notification_service = NotificationService()
        self.offline_sync = OfflineSyncManager()
        
    def process_location_update(self, partner_id: str, location_data: Dict) -> Dict:
        """
        Process real-time location update from delivery partner
        """
        
        # Validate and normalize location data
        validated_location = self.location_processor.validate_location(location_data)
        
        if not validated_location['valid']:
            return {'status': 'error', 'message': 'Invalid location data'}
        
        # Get current active deliveries for partner
        active_deliveries = self._get_active_deliveries(partner_id)
        
        # Update location and calculate ETAs for each delivery
        updates = []
        for delivery in active_deliveries:
            
            # Calculate new ETA
            new_eta = self.eta_calculator.calculate_eta(
                current_location=validated_location['location'],
                destination=delivery['customer_location'],
                traffic_conditions=validated_location['traffic_context']
            )
            
            # Check if significant ETA change (>2 minutes difference)
            if abs(new_eta - delivery['current_eta']) > 2:
                
                # Send update to customer
                customer_update = {
                    'order_id': delivery['order_id'],
                    'partner_location': validated_location['location'],
                    'new_eta_minutes': new_eta,
                    'status': delivery['status'],
                    'timestamp': datetime.utcnow().isoformat()
                }
                
                self.notification_service.send_real_time_update(
                    delivery['customer_id'], customer_update
                )
                
                updates.append(customer_update)
        
        return {
            'status': 'success',
            'updates_sent': len(updates),
            'partner_id': partner_id
        }

class SwiggyRestaurantDiscoverySystem:
    def __init__(self):
        self.restaurants = {}  # restaurant_id -> RestaurantInfo
        self.cuisine_index = defaultdict(set)  # cuisine -> set of restaurant_ids
        self.location_index = {}  # area -> list of restaurant_ids  
        self.menu_search_index = {}  # normalized_item_name -> list of (restaurant_id, item_id)
        self.user_profiles = {}  # user_id -> UserProfile
        
        # Search optimization structures
        self.restaurant_name_trie = TrieNode()
        self.menu_item_trie = TrieNode()
        
        # Recommendation system
        self.recommendation_engine = RecommendationEngine()
        
    def add_restaurant(self, restaurant: RestaurantInfo) -> bool:
        """
        Add restaurant to the discovery system with full indexing
        """
        try:
            # Store restaurant
            self.restaurants[restaurant.restaurant_id] = restaurant
            
            # Index by cuisine types
            for cuisine in restaurant.cuisine_types:
                self.cuisine_index[cuisine].add(restaurant.restaurant_id)
            
            # Index by location/area
            if restaurant.area not in self.location_index:
                self.location_index[restaurant.area] = []
            self.location_index[restaurant.area].append(restaurant.restaurant_id)
            
            # Index restaurant name in trie
            self._add_to_trie(self.restaurant_name_trie, restaurant.name.lower(), 
                            'restaurant', restaurant.restaurant_id)
            
            # Index menu items
            for item in restaurant.menu_items:
                # Add to menu search index
                normalized_name = self._normalize_text(item.name)
                if normalized_name not in self.menu_search_index:
                    self.menu_search_index[normalized_name] = []
                self.menu_search_index[normalized_name].append((restaurant.restaurant_id, item.item_id))
                
                # Add to menu item trie
                self._add_to_trie(self.menu_item_trie, item.name.lower(), 
                                'menu_item', (restaurant.restaurant_id, item.item_id))
            
            return True
            
        except Exception as e:
            print(f"Error adding restaurant: {e}")
            return False
    
    def search_restaurants(self, query: str, user_id: str, 
                         filters: SearchFilters, limit: int = 20) -> List[Dict]:
        """
        Search restaurants with advanced filtering and personalization
        """
        
        user_profile = self.user_profiles.get(user_id)
        if not user_profile:
            user_profile = UserProfile(user_id, {'lat': 0, 'lng': 0})
        
        # Step 1: Get candidate restaurants based on query
        candidate_restaurants = self._get_search_candidates(query, user_profile)
        
        # Step 2: Apply filters
        filtered_restaurants = self._apply_filters(candidate_restaurants, filters, user_profile)
        
        # Step 3: Score and rank restaurants
        scored_restaurants = self._score_restaurants(filtered_restaurants, query, user_profile)
        
        # Step 4: Sort by score and return top results
        scored_restaurants.sort(key=lambda x: x['score'], reverse=True)
        
        # Step 5: Format results
        results = []
        for scored_restaurant in scored_restaurants[:limit]:
            restaurant = self.restaurants[scored_restaurant['restaurant_id']]
            
            result = {
                'restaurant_id': restaurant.restaurant_id,
                'name': restaurant.name,
                'cuisine_types': [c.value for c in restaurant.cuisine_types],
                'rating': restaurant.rating,
                'price_range': restaurant.price_range.value,
                'delivery_time_minutes': restaurant.delivery_time_minutes,
                'distance_km': self._calculate_distance(user_profile.location, restaurant.location),
                'offers': restaurant.offers,
                'image_url': restaurant.image_url,
                'relevance_score': scored_restaurant['score'],
                'match_reasons': scored_restaurant.get('match_reasons', [])
            }
            results.append(result)
        
        return results
    
    def _get_search_candidates(self, query: str, user_profile: UserProfile) -> Set[str]:
        """
        Get initial candidate restaurants based on search query
        """
        candidates = set()
        
        if not query or query.strip() == "":
            # No specific query - return restaurants near user
            candidates = self._get_nearby_restaurants(user_profile)
        else:
            normalized_query = self._normalize_text(query)
            
            # Search restaurant names
            restaurant_matches = self._search_trie(self.restaurant_name_trie, normalized_query.lower())
            for match in restaurant_matches:
                if match['type'] == 'restaurant':
                    candidates.add(match['id'])
            
            # Search menu items
            menu_matches = self._search_trie(self.menu_item_trie, normalized_query.lower())
            for match in menu_matches:
                if match['type'] == 'menu_item':
                    candidates.add(match['id'][0])  # restaurant_id is first element of tuple
            
            # Search by cuisine name
            cuisine_matches = self._search_by_cuisine_name(normalized_query)
            candidates.update(cuisine_matches)
            
            # Fuzzy search for common typos
            fuzzy_matches = self._fuzzy_search(normalized_query)
            candidates.update(fuzzy_matches)
        
        return candidates
    
    def _apply_filters(self, candidates: Set[str], filters: SearchFilters, 
                      user_profile: UserProfile) -> List[str]:
        """
        Apply search filters to candidate restaurants
        """
        filtered = []
        
        for restaurant_id in candidates:
            restaurant = self.restaurants[restaurant_id]
            
            # Check if restaurant is open
            if not restaurant.is_open:
                continue
            
            # Cuisine filter
            if filters.cuisine_types:
                if not any(cuisine in restaurant.cuisine_types for cuisine in filters.cuisine_types):
                    continue
            
            # Price range filter
            if filters.price_ranges:
                if restaurant.price_range not in filters.price_ranges:
                    continue
            
            # Delivery time filter
            if filters.max_delivery_time:
                if restaurant.delivery_time_minutes > filters.max_delivery_time:
                    continue
            
            # Rating filter
            if filters.min_rating:
                if restaurant.rating < filters.min_rating:
                    continue
            
            # Vegetarian filter
            if filters.is_veg_only:
                has_veg_options = any(item.is_veg for item in restaurant.menu_items)
                if not has_veg_options:
                    continue
            
            # Offers filter
            if filters.has_offers:
                if not restaurant.offers:
                    continue
            
            # Distance filter
            if filters.max_distance_km:
                distance = self._calculate_distance(user_profile.location, restaurant.location)
                if distance > filters.max_distance_km:
                    continue
            
            filtered.append(restaurant_id)
        
        return filtered
    
    def _score_restaurants(self, restaurant_ids: List[str], query: str, 
                          user_profile: UserProfile) -> List[Dict]:
        """
        Score restaurants for ranking using multiple factors
        """
        scored_restaurants = []
        
        for restaurant_id in restaurant_ids:
            restaurant = self.restaurants[restaurant_id]
            
            # Calculate various scoring factors
            scores = {
                'relevance': self._calculate_relevance_score(restaurant, query),
                'rating': restaurant.rating / 5.0,  # Normalize to 0-1
                'delivery_time': max(0, 1.0 - (restaurant.delivery_time_minutes / 60.0)),
                'distance': self._calculate_distance_score(user_profile.location, restaurant.location),
                'personalization': self._calculate_personalization_score(restaurant, user_profile),
                'popularity': self._calculate_popularity_score(restaurant),
                'offers': 0.1 if restaurant.offers else 0.0
            }
            
            # Weighted final score
            weights = {
                'relevance': 0.25,
                'rating': 0.20,
                'delivery_time': 0.15,
                'distance': 0.15,
                'personalization': 0.15,
                'popularity': 0.05,
                'offers': 0.05
            }
            
            final_score = sum(scores[factor] * weights[factor] for factor in scores)
            
            # Generate match reasons for transparency
            match_reasons = self._generate_match_reasons(restaurant, query, scores)
            
            scored_restaurants.append({
                'restaurant_id': restaurant_id,
                'score': final_score,
                'score_breakdown': scores,
                'match_reasons': match_reasons
            })
        
        return scored_restaurants
    
    def _calculate_relevance_score(self, restaurant: RestaurantInfo, query: str) -> float:
        """
        Calculate how relevant restaurant is to search query
        """
        if not query:
            return 0.5  # Neutral score for empty query
        
        normalized_query = self._normalize_text(query).lower()
        score = 0.0
        
        # Restaurant name match
        restaurant_name = restaurant.name.lower()
        if normalized_query in restaurant_name:
            score += 0.4
            if restaurant_name.startswith(normalized_query):
                score += 0.2  # Bonus for name starting with query
        
        # Cuisine match
        for cuisine in restaurant.cuisine_types:
            if normalized_query in cuisine.value.replace('_', ' '):
                score += 0.3
        
        # Menu item match
        for item in restaurant.menu_items:
            if normalized_query in item.name.lower():
                score += 0.2
                break  # Don't double count menu matches
        
        return min(1.0, score)
    
    def _calculate_personalization_score(self, restaurant: RestaurantInfo, 
                                       user_profile: UserProfile) -> float:
        """
        Calculate personalization score based on user history and preferences
        """
        score = 0.0
        
        # Cuisine preference match
        if user_profile.preferred_cuisines:
            common_cuisines = set(restaurant.cuisine_types) & set(user_profile.preferred_cuisines)
            score += len(common_cuisines) / len(user_profile.preferred_cuisines) * 0.4
        
        # Order history match
        if restaurant.restaurant_id in user_profile.order_history:
            score += 0.3  # Bonus for previously ordered restaurant
        
        # Price range match with user's average order value
        price_match_score = self._calculate_price_match_score(restaurant, user_profile)
        score += price_match_score * 0.3
        
        return min(1.0, score)
    
    def get_personalized_recommendations(self, user_id: str, 
                                       limit: int = 10) -> List[Dict]:
        """
        Get personalized restaurant recommendations for user
        """
        user_profile = self.user_profiles.get(user_id)
        if not user_profile:
            return self._get_popular_restaurants(limit)
        
        # Use recommendation engine for collaborative filtering
        recommendations = self.recommendation_engine.get_recommendations(
            user_id, user_profile, self.restaurants, limit
        )
        
        return recommendations
    
    def update_user_interaction(self, user_id: str, restaurant_id: str, 
                              interaction_type: str) -> bool:
        """
        Update user profile based on interactions (order, view, search)
        """
        try:
            if user_id not in self.user_profiles:
                self.user_profiles[user_id] = UserProfile(user_id, {'lat': 0, 'lng': 0})
            
            user_profile = self.user_profiles[user_id]
            
            if interaction_type == 'order' and restaurant_id in self.restaurants:
                # Add to order history
                user_profile.order_history.append(restaurant_id)
                
                # Update preferred cuisines
                restaurant = self.restaurants[restaurant_id]
                for cuisine in restaurant.cuisine_types:
                    if cuisine not in user_profile.preferred_cuisines:
                        user_profile.preferred_cuisines.append(cuisine)
                
                # Keep only last 50 orders for relevance
                if len(user_profile.order_history) > 50:
                    user_profile.order_history = user_profile.order_history[-50:]
            
            return True
            
        except Exception as e:
            print(f"Error updating user interaction: {e}")
            return False
    
    # Helper methods
    def _normalize_text(self, text: str) -> str:
        """Normalize text for search"""
        # Remove special characters, convert to lowercase
        return re.sub(r'[^\w\s]', '', text.lower().strip())
    
    def _calculate_distance(self, loc1: Dict, loc2: Dict) -> float:
        """Calculate distance between two locations (simplified)"""
        import math
        
        lat1, lng1 = loc1.get('lat', 0), loc1.get('lng', 0)
        lat2, lng2 = loc2.get('lat', 0), loc2.get('lng', 0)
        
        # Haversine formula (simplified for interview)
        dlat = math.radians(lat2 - lat1)
        dlng = math.radians(lng2 - lng1)
        
        a = (math.sin(dlat/2)**2 + 
             math.cos(math.radians(lat1)) * math.cos(math.radians(lat2)) * 
             math.sin(dlng/2)**2)
        
        c = 2 * math.asin(math.sqrt(a))
        return 6371 * c  # Earth's radius in km
    
    def _calculate_distance_score(self, user_loc: Dict, restaurant_loc: Dict) -> float:
        """Calculate distance-based score (closer = higher score)"""
        distance = self._calculate_distance(user_loc, restaurant_loc)
        return max(0, 1.0 - (distance / 10.0))  # Score decreases with distance
    
    def _add_to_trie(self, root: 'TrieNode', word: str, data_type: str, data_id: any):
        """Add word to trie structure"""
        node = root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        
        node.is_end_word = True
        node.data.append({'type': data_type, 'id': data_id})

class RecommendationEngine:
    def get_recommendations(self, user_id: str, user_profile: UserProfile, 
                          restaurants: Dict, limit: int) -> List[Dict]:
        """Simple collaborative filtering recommendations"""
        # Simplified implementation for interview
        recommended = []
        
        # Recommend based on cuisine preferences
        for cuisine in user_profile.preferred_cuisines:
            for restaurant_id, restaurant in restaurants.items():
                if (cuisine in restaurant.cuisine_types and 
                    restaurant_id not in user_profile.order_history and
                    len(recommended) < limit):
                    
                    recommended.append({
                        'restaurant_id': restaurant_id,
                        'name': restaurant.name,
                        'reason': f'Based on your interest in {cuisine.value} cuisine'
                    })
        
        return recommended[:limit]
