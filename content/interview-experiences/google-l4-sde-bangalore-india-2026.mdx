```python
from typing import List, Dict, Optional, Set
from dataclasses import dataclass
from enum import Enum
import time
import hashlib

class MessageStatus(Enum):
    SENT = "sent"
    DELIVERED = "delivered"
    READ = "read"
    FAILED = "failed"

class NetworkQuality(Enum):
    HIGH = "4G/5G"
    MEDIUM = "3G"
    LOW = "2G/EDGE"
    OFFLINE = "offline"

@dataclass
class Message:
    message_id: str
    sender_id: str
    recipient_id: str
    content: str
    timestamp: int
    message_type: str  # text, image, audio, video
    language: str = "en"  # Support for Indian languages
    
@dataclass
class UserDevice:
    user_id: str
    device_id: str
    network_quality: NetworkQuality
    last_seen: int
    preferred_language: str
    data_saver_mode: bool = False  # For cost-conscious Indian users

class IndianMessagingSystem:
    def __init__(self):
        self.message_queue = {}  # user_id -> List[Message]
        self.delivery_status = {}  # message_id -> MessageStatus
        self.user_devices = {}  # user_id -> UserDevice
        self.offline_storage = {}  # user_id -> List[Message]
        
    def send_message(self, message: Message) -> bool:
        """
        Send message with India-specific optimizations
        """
        try:
            # Generate unique message ID
            message.message_id = self._generate_message_id(message)
            
            # Check if recipient is online
            recipient_device = self.user_devices.get(message.recipient_id)
            
            if recipient_device and self._is_user_online(recipient_device):
                # Direct delivery for online users
                delivery_success = self._deliver_message_direct(message, recipient_device)
                if delivery_success:
                    self.delivery_status[message.message_id] = MessageStatus.DELIVERED
                    return True
            
            # Store for offline delivery
            self._store_for_offline_delivery(message)
            self.delivery_status[message.message_id] = MessageStatus.SENT
            
            return True
            
        except Exception as e:
            print(f"Error sending message: {e}")
            self.delivery_status[message.message_id] = MessageStatus.FAILED
            return False
    
    def _deliver_message_direct(self, message: Message, device: UserDevice) -> bool:
        """
        Deliver message with network quality optimization
        """
        # Compress message based on network quality
        compressed_message = self._compress_for_network(message, device.network_quality)
        
        # Use data saver optimizations for cost-conscious users
        if device.data_saver_mode:
            compressed_message = self._apply_data_saver_optimizations(compressed_message)
        
        # Simulate network delivery
        if device.network_quality == NetworkQuality.LOW:
            # For 2G networks, use SMS fallback for critical messages
            return self._deliver_via_sms_fallback(compressed_message)
        else:
            # Standard internet delivery
            return self._deliver_via_internet(compressed_message, device)
    
    def _compress_for_network(self, message: Message, network_quality: NetworkQuality) -> Message:
        """
        Apply network-specific compression
        """
        if network_quality == NetworkQuality.LOW:
            # Aggressive compression for 2G
            if message.message_type == "image":
                # Convert to low-quality JPEG
                message.content = self._compress_image_aggressive(message.content)
            elif message.message_type == "video":
                # Convert to audio-only or thumbnail
                message.content = self._extract_audio_or_thumbnail(message.content)
                message.message_type = "audio"  # Downgrade video to audio
                
        elif network_quality == NetworkQuality.MEDIUM:
            # Moderate compression for 3G
            if message.message_type == "image":
                message.content = self._compress_image_moderate(message.content)
                
        return message
    
    def _apply_data_saver_optimizations(self, message: Message) -> Message:
        """
        Apply optimizations for users concerned about data costs
        """
        # Don't auto-download media content
        if message.message_type in ["image", "video", "audio"]:
            # Send metadata only, actual media downloaded on user request
            original_content = message.content
            message.content = {
                "type": "media_placeholder",
                "size": len(original_content),
                "download_url": self._generate_download_url(message.message_id),
                "preview": self._generate_preview(original_content)
            }
        
        return message
    
    def _deliver_via_sms_fallback(self, message: Message) -> bool:
        """
        SMS fallback for extremely poor network conditions
        """
        if message.message_type == "text" and len(message.content) <= 160:
            # Use SMS API for text messages under 160 characters
            return self._send_sms(message.recipient_id, message.content)
        else:
            # Store for later delivery when network improves
            self._store_for_offline_delivery(message)
            return False
    
    def handle_user_online(self, user_id: str):
        """
        Handle user coming online - deliver pending messages
        """
        if user_id in self.offline_storage:
            pending_messages = self.offline_storage[user_id]
            device = self.user_devices.get(user_id)
            
            if device:
                # Deliver messages based on network quality and priority
                for message in sorted(pending_messages, key=lambda m: m.timestamp):
                    if self._deliver_message_direct(message, device):
                        self.delivery_status[message.message_id] = MessageStatus.DELIVERED
                        pending_messages.remove(message)
                
                # Update offline storage
                if not pending_messages:
                    del self.offline_storage[user_id]
                else:
                    self.offline_storage[user_id] = pending_messages
    
    def get_message_status(self, message_id: str) -> MessageStatus:
        """
        Get current message delivery status
        """
        return self.delivery_status.get(message_id, MessageStatus.FAILED)
    
    def _generate_message_id(self, message: Message) -> str:
        """
        Generate unique message ID
        """
        data = f"{message.sender_id}{message.recipient_id}{message.timestamp}{message.content}"
        return hashlib.md5(data.encode()).hexdigest()
    
    def _is_user_online(self, device: UserDevice) -> bool:
        """
        Check if user is currently online
        """
        # Consider user online if last seen within 30 seconds
        return (time.time() - device.last_seen) < 30
    
    # Additional helper methods for compression, SMS, etc.
    def _compress_image_aggressive(self, content: str) -> str:
        # Simulate aggressive image compression
        return content[:len(content)//4]  # 75% compression
    
    def _compress_image_moderate(self, content: str) -> str:
        # Simulate moderate image compression  
        return content[:len(content)//2]  # 50% compression
    
    def _send_sms(self, phone_number: str, content: str) -> bool:
        # Simulate SMS API call
        print(f"Sending SMS to {phone_number}: {content}")
        return True
    
    def _store_for_offline_delivery(self, message: Message):
        """
        Store message for offline delivery
        """
        if message.recipient_id not in self.offline_storage:
            self.offline_storage[message.recipient_id] = []
        self.offline_storage[message.recipient_id].append(message)
```

**Context**: Build a seat reservation system for Indian Railways that can handle festival season booking rush (millions of concurrent users).

**Requirements**:

```python
from typing import Dict, List, Optional, Set
from dataclasses import dataclass
from enum import Enum
import heapq
from datetime import datetime, timedelta

class SeatClass(Enum):
    SLEEPER = "SL"
    AC_3_TIER = "3A"
    AC_2_TIER = "2A"
    FIRST_CLASS = "1A"

class SeatStatus(Enum):
    AVAILABLE = "available"
    CONFIRMED = "confirmed"
    RAC = "rac"
    WAITLISTED = "waitlisted"

class QuotaType(Enum):
    GENERAL = "GN"
    LADIES = "LD"
    SENIOR_CITIZEN = "SS"
    TATKAL = "TQ"
    PREMIUM_TATKAL = "PT"

@dataclass
class Seat:
    coach_number: str
    seat_number: str
    seat_class: SeatClass
    status: SeatStatus = SeatStatus.AVAILABLE
    passenger_id: Optional[str] = None

@dataclass
class Passenger:
    passenger_id: str
    name: str
    age: int
    gender: str
    phone_number: str
    
    def get_quota_eligibility(self) -> List[QuotaType]:
        """Determine which quotas this passenger is eligible for"""
        eligible_quotas = [QuotaType.GENERAL]
        
        if self.gender == "F":
            eligible_quotas.append(QuotaType.LADIES)
        
        if self.age >= 60:
            eligible_quotas.append(QuotaType.SENIOR_CITIZEN)
            
        return eligible_quotas

@dataclass
class BookingRequest:
    request_id: str
    train_number: str
    journey_date: str
    source_station: str
    destination_station: str
    passengers: List[Passenger]
    preferred_class: SeatClass
    quota: QuotaType = QuotaType.GENERAL

class TrainReservationSystem:
    def __init__(self):
        self.trains = {}  # train_number -> Train configuration
        self.seat_matrix = {}  # (train_number, date, class) -> List[Seat]
        self.quota_limits = {}  # (train_number, class, quota) -> limit
        self.waitlist_queues = {}  # (train_number, date, class) -> priority queue
        self.rac_queues = {}  # (train_number, date, class) -> RAC queue
        self.bookings = {}  # booking_id -> BookingRequest
        
        # Initialize quota limits (typical Indian Railways distribution)
        self._initialize_quota_limits()
    
    def _initialize_quota_limits(self):
        """Initialize typical quota distributions for Indian Railways"""
        # Typical quota distribution for a train
        quota_percentages = {
            QuotaType.GENERAL: 0.65,  # 65% general quota
            QuotaType.LADIES: 0.10,   # 10% ladies quota  
            QuotaType.SENIOR_CITIZEN: 0.05,  # 5% senior citizen
            QuotaType.TATKAL: 0.20    # 20% tatkal quota
        }
        
        # This would be initialized per train/class combination
        for train_number in ["12345", "67890"]:  # Example trains
            for seat_class in SeatClass:
                total_seats = self._get_total_seats(train_number, seat_class)
                for quota, percentage in quota_percentages.items():
                    key = (train_number, seat_class, quota)
                    self.quota_limits[key] = int(total_seats * percentage)
    
    def book_ticket(self, booking_request: BookingRequest) -> Dict[str, str]:
        """
        Main booking function handling the complex Indian Railways logic
        """
        try:
            # Validate booking request
            if not self._validate_booking_request(booking_request):
                return {"status": "INVALID", "message": "Invalid booking request"}
            
            # Check passenger quota eligibility
            quota_results = {}
            
            for passenger in booking_request.passengers:
                result = self._book_single_passenger(
                    booking_request, passenger, booking_request.preferred_class
                )
                quota_results[passenger.passenger_id] = result
            
            # Generate PNR and booking confirmation
            pnr = self._generate_pnr()
            self.bookings[pnr] = booking_request
            
            return {
                "status": "SUCCESS",
                "pnr": pnr,
                "passenger_results": quota_results,
                "booking_date": datetime.now().isoformat()
            }
            
        except Exception as e:
            return {"status": "ERROR", "message": str(e)}
    
    def _book_single_passenger(self, booking_request: BookingRequest, 
                              passenger: Passenger, seat_class: SeatClass) -> Dict:
        """
        Book seat for individual passenger with quota logic
        """
        eligible_quotas = passenger.get_quota_eligibility()
        
        # Try booking in order of quota preference
        for quota in eligible_quotas:
            result = self._try_booking_in_quota(
                booking_request, passenger, seat_class, quota
            )
            
            if result["status"] != "NO_AVAILABILITY":
                return result
        
        # If no quota has availability, add to general waitlist
        return self._add_to_waitlist(booking_request, passenger, seat_class)
    
    def _try_booking_in_quota(self, booking_request: BookingRequest,
                             passenger: Passenger, seat_class: SeatClass,
                             quota: QuotaType) -> Dict:
        """
        Try to book in specific quota
        """
        train_number = booking_request.train_number
        journey_date = booking_request.journey_date
        
        # Check quota availability
        quota_key = (train_number, seat_class, quota)
        available_seats = self._get_available_seats_in_quota(quota_key, journey_date)
        
        if available_seats:
            # Confirm booking
            seat = available_seats[0]
            seat.status = SeatStatus.CONFIRMED
            seat.passenger_id = passenger.passenger_id
            
            return {
                "status": "CONFIRMED",
                "seat_number": f"{seat.coach_number}/{seat.seat_number}",
                "quota": quota.value,
                "class": seat_class.value
            }
        
        # Check RAC availability
        rac_seats = self._get_rac_availability(quota_key, journey_date)
        if rac_seats:
            return self._assign_rac(passenger, seat_class, quota, journey_date)
        
        return {"status": "NO_AVAILABILITY"}
    
    def _assign_rac(self, passenger: Passenger, seat_class: SeatClass,
                   quota: QuotaType, journey_date: str) -> Dict:
        """
        Assign RAC (Reservation Against Cancellation) berth
        """
        # RAC logic: 2 passengers share 1 berth until confirmation
        rac_key = (passenger, seat_class, quota, journey_date)
        
        if not hasattr(self, 'rac_assignments'):
            self.rac_assignments = {}
        
        # Find RAC berth
        rac_berth_id = self._allocate_rac_berth(seat_class, journey_date)
        
        return {
            "status": "RAC",
            "rac_number": rac_berth_id,
            "quota": quota.value,
            "class": seat_class.value,
            "message": "RAC - berth sharing until confirmation"
        }
    
    def _add_to_waitlist(self, booking_request: BookingRequest,
                        passenger: Passenger, seat_class: SeatClass) -> Dict:
        """
        Add passenger to waitlist with priority queue
        """
        waitlist_key = (booking_request.train_number, booking_request.journey_date, seat_class)
        
        if waitlist_key not in self.waitlist_queues:
            self.waitlist_queues[waitlist_key] = []
        
        # Priority based on booking time and passenger category
        priority = self._calculate_waitlist_priority(passenger)
        
        heapq.heappush(self.waitlist_queues[waitlist_key], (priority, passenger.passenger_id))
        
        waitlist_position = len(self.waitlist_queues[waitlist_key])
        
        return {
            "status": "WAITLISTED",
            "waitlist_number": f"WL{waitlist_position}",
            "class": seat_class.value,
            "message": f"Waitlist position {waitlist_position}"
        }
    
    def handle_cancellation(self, pnr: str, passenger_id: str):
        """
        Handle seat cancellation and promote waitlisted passengers
        """
        if pnr not in self.bookings:
            return False
        
        booking = self.bookings[pnr]
        
        # Find and free the seat
        freed_seat = self._free_passenger_seat(booking, passenger_id)
        
        if freed_seat:
            # Promote from RAC first, then waitlist
            self._promote_from_rac_or_waitlist(
                booking.train_number, booking.journey_date, freed_seat.seat_class
            )
        
        return True
    
    def _promote_from_rac_or_waitlist(self, train_number: str, 
                                    journey_date: str, seat_class: SeatClass):
        """
        Promote passengers from RAC to confirmed, or waitlist to RAC
        """
        # First promote RAC passengers to confirmed
        rac_key = (train_number, journey_date, seat_class)
        if rac_key in self.rac_queues and self.rac_queues[rac_key]:
            rac_passenger = self.rac_queues[rac_key].pop(0)
            # Promote to confirmed seat
            self._confirm_rac_passenger(rac_passenger, seat_class)
        
        # Then promote waitlist to RAC  
        waitlist_key = (train_number, journey_date, seat_class)
        if waitlist_key in self.waitlist_queues and self.waitlist_queues[waitlist_key]:
            _, waitlisted_passenger_id = heapq.heappop(self.waitlist_queues[waitlist_key])
            # Move to RAC
            self._move_waitlist_to_rac(waitlisted_passenger_id, seat_class, journey_date)
    
    def check_booking_status(self, pnr: str) -> Dict:
        """
        Check current status of booking (for live status updates)
        """
        if pnr not in self.bookings:
            return {"error": "Invalid PNR"}
        
        booking = self.bookings[pnr]
        passenger_statuses = []
        
        for passenger in booking.passengers:
            status = self._get_current_passenger_status(passenger.passenger_id, booking)
            passenger_statuses.append({
                "name": passenger.name,
                "status": status["status"],
                "seat_info": status.get("seat_info", ""),
                "current_status": status.get("current_status", "")
            })
        
        return {
            "pnr": pnr,
            "train_number": booking.train_number,
            "journey_date": booking.journey_date,
            "passengers": passenger_statuses
        }
    
    # Helper methods for quota limits, seat allocation, etc.
    def _get_total_seats(self, train_number: str, seat_class: SeatClass) -> int:
        # Return total seats for given train and class
        seat_counts = {
            SeatClass.SLEEPER: 72,    # Typical sleeper coach
            SeatClass.AC_3_TIER: 64,  # Typical 3AC coach
            SeatClass.AC_2_TIER: 46,  # Typical 2AC coach  
            SeatClass.FIRST_CLASS: 18 # Typical First AC coach
        }
        return seat_counts.get(seat_class, 50)
    
    def _validate_booking_request(self, request: BookingRequest) -> bool:
        # Validate train exists, date is valid, etc.
        return True
    
    def _generate_pnr(self) -> str:
        # Generate 10-digit PNR
        import random
        return str(random.randint(1000000000, 9999999999))
    
    def _get_available_seats_in_quota(self, quota_key, journey_date) -> List[Seat]:
        # Return available seats for specific quota
        return []  # Simplified for interview
    
    def _calculate_waitlist_priority(self, passenger: Passenger) -> int:
        # Higher priority (lower number) for senior citizens, ladies, etc.
        priority = 100
        if passenger.age >= 60:
            priority -= 20
        if passenger.gender == "F":
            priority -= 10
        return priority
```

**My Dynamic Programming Solution**:

```python

from dataclasses import dataclass
from enum import Enum

class PlayerRole(Enum):

    BATSMAN = "BAT"

@dataclass

    def get_value_ratio(self) -> float:
        """Points per lakh ratio for optimization"""
        return self.expected_points / self.price if self.price > 0 else 0

class FantasyTeamSelector:
    def __init__(self, players: List[Player], budget: int = 10000):  # 100 crores in lakhs
        self.players = players
        self.budget = budget

        """
        Use dynamic programming to find optimal team selection
        """
        # Group players by role for easier processing

        # Generate all valid team compositions
        valid_compositions = self._generate_valid_compositions()
        
        best_team = None
        best_points = 0
        
        # Try each valid composition
        for composition in valid_compositions:
            team = self._optimize_for_composition(players_by_role, composition)
            if team and team['total_points'] > best_points:
                best_points = team['total_points']
                best_team = team
        
        return best_team
    
    def _group_players_by_role(self) -> Dict[PlayerRole, List[Player]]:
        """Group players by their roles"""
        grouped = {role: [] for role in PlayerRole}
        
        for player in self.players:
            grouped[player.role].append(player)
        
        # Sort by value ratio (points per price) descending
        for role in grouped:
            grouped[role].sort(key=lambda p: p.get_value_ratio(), reverse=True)
        
        return grouped
    
    def _generate_valid_compositions(self) -> List[Dict[PlayerRole, int]]:
        """Generate all valid team compositions within constraints"""
        compositions = []
        
        # Iterate through all possible combinations
        for wk in range(self.team_constraints[PlayerRole.WICKETKEEPER][0], 
                       self.team_constraints[PlayerRole.WICKETKEEPER][1] + 1):
            for bat in range(self.team_constraints[PlayerRole.BATSMAN][0], 
                           self.team_constraints[PlayerRole.BATSMAN][1] + 1):
                for ar in range(self.team_constraints[PlayerRole.ALL_ROUNDER][0], 
                              self.team_constraints[PlayerRole.ALL_ROUNDER][1] + 1):
                    for bowl in range(self.team_constraints[PlayerRole.BOWLER][0], 
                                    self.team_constraints[PlayerRole.BOWLER][1] + 1):
                        
                        if wk + bat + ar + bowl == self.total_players:
                            compositions.append({
                                PlayerRole.WICKETKEEPER: wk,
                                PlayerRole.BATSMAN: bat,
                                PlayerRole.ALL_ROUNDER: ar,
                                PlayerRole.BOWLER: bowl
                            })
        
        return compositions
    
    def _optimize_for_composition(self, players_by_role: Dict[PlayerRole, List[Player]], 
                                composition: Dict[PlayerRole, int]) -> Dict:
        """
        Use DP to optimize team selection for given composition
        """
        # This is a complex multi-dimensional knapsack problem
        # We'll use a greedy approach with backtracking for the interview
        
        selected_team = []
        total_cost = 0
        total_points = 0
        team_count = {}  # Count players per IPL team
        
        # Select players for each role
        for role, count_needed in composition.items():
            role_players = self._select_role_players(
                players_by_role[role], count_needed, 
                self.budget - total_cost, team_count
            )
            
            if len(role_players) < count_needed:
                return None  # Cannot satisfy constraints
            
            selected_team.extend(role_players)
            total_cost += sum(p.price for p in role_players)
            total_points += sum(p.expected_points for p in role_players)
            
            # Update team count
            for player in role_players:
                team_count[player.team] = team_count.get(player.team, 0) + 1
        
        # Validate team constraints
        if not self._validate_team_constraints(selected_team, team_count):
            return None
        
        return {
            'players': selected_team,
            'total_cost': total_cost,
            'total_points': total_points,
            'remaining_budget': self.budget - total_cost,
            'team_composition': composition,
            'team_distribution': team_count
        }
    
    def _select_role_players(self, role_players: List[Player], count_needed: int, 
                           remaining_budget: int, current_team_count: Dict[str, int]) -> List[Player]:
        """
        Select best players for a role within constraints using DP approach
        """
        # Filter players that fit budget and team constraints
        eligible_players = []
        
        for player in role_players:
            if player.price <= remaining_budget:
                team_count = current_team_count.get(player.team, 0)
                if team_count < self.max_players_per_team:
                    eligible_players.append(player)
        
        if len(eligible_players) < count_needed:
            return eligible_players  # Return what we can get
        
        # Use DP to find best combination
        # For interview simplicity, using greedy with value ratio
        selected = []
        used_budget = 0
        temp_team_count = current_team_count.copy()
        
        for player in eligible_players:
            if len(selected) >= count_needed:
                break
            
            if (used_budget + player.price <= remaining_budget and 
                temp_team_count.get(player.team, 0) < self.max_players_per_team):
                
                selected.append(player)
                used_budget += player.price
                temp_team_count[player.team] = temp_team_count.get(player.team, 0) + 1
        
        return selected
    
    def _validate_team_constraints(self, team: List[Player], team_count: Dict[str, int]) -> bool:
        """Validate all team constraints are satisfied"""
        
        # Check team size
        if len(team) != self.total_players:
            return False
        
        # Check role constraints
        role_count = {}
        for player in team:
            role_count[player.role] = role_count.get(player.role, 0) + 1
        
        for role, (min_count, max_count) in self.team_constraints.items():
            actual_count = role_count.get(role, 0)
            if actual_count < min_count or actual_count > max_count:
                return False
        
        # Check IPL team constraint (max 7 from one team)
        for team_name, count in team_count.items():
            if count > self.max_players_per_team:
                return False
        
        return True
```
