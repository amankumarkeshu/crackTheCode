class SimpleInventoryManager(InventoryManager):
    def __init__(self):
        self.inventory = {}  # product_id -> available_quantity
        self.reserved = {}   # product_id -> reserved_quantity
        self.lock = threading.RLock()
    
    def initialize_inventory(self, products: List[Product]):
        """Initialize inventory with products"""
        with self.lock:
            for product in products:
                self.inventory[product.product_id] = product.available_quantity
                self.reserved[product.product_id] = 0
    
    def check_availability(self, product_id: str, quantity: int) -> bool:
        with self.lock:
            available = self.inventory.get(product_id, 0)
            reserved = self.reserved.get(product_id, 0)
            return (available - reserved) >= quantity
    
    def reserve_inventory(self, product_id: str, quantity: int) -> bool:
        with self.lock:
            if self.check_availability(product_id, quantity):
                self.reserved[product_id] = self.reserved.get(product_id, 0) + quantity
                return True
            return False
    
    def release_inventory(self, product_id: str, quantity: int) -> bool:
        with self.lock:
            current_reserved = self.reserved.get(product_id, 0)
            if current_reserved >= quantity:
                self.reserved[product_id] = current_reserved - quantity
                return True
            return False

    def create_cart(self, user_id: str) -> str:
        """Create new cart for user"""
        cart_id = f"CART_{uuid.uuid4().hex[:8].upper()}"
        
        with self.lock:

    def add_item_to_cart(self, cart_id: str, product_id: str, quantity: int) -> Dict[str, any]:
        """Add item to cart with inventory validation"""
        
        # Validate inputs
        if quantity <= 0:
            return self._error_response("Quantity must be positive")
        
        if cart_id not in self.carts:
            return self._error_response("Cart not found")
        
        if product_id not in self.products:
            return self._error_response("Product not found")
        
        cart = self.carts[cart_id]
        product = self.products[product_id]
        
        # Check cart status
        if cart.status != CartStatus.ACTIVE:
            return self._error_response("Cart is not active")
        
        with self.lock:
            # Calculate total quantity needed (existing + new)
            existing_quantity = 0
            if product_id in cart.items:
                existing_quantity = cart.items[product_id].quantity
                # Release existing reservation
                self.inventory_manager.release_inventory(product_id, existing_quantity)
            
            total_quantity = existing_quantity + quantity
            
            # Check inventory availability
            if not self.inventory_manager.check_availability(product_id, total_quantity):
                # Re-reserve existing quantity if it was released
                if existing_quantity > 0:
                    self.inventory_manager.reserve_inventory(product_id, existing_quantity)
                return self._error_response(f"Insufficient inventory. Available: {self._get_available_quantity(product_id)}")
            
            # Reserve inventory
            if not self.inventory_manager.reserve_inventory(product_id, total_quantity):
                return self._error_response("Failed to reserve inventory")
            
            # Add/Update cart item
            cart.items[product_id] = CartItem(product, total_quantity)
            cart.update_timestamp()
            
            return self._success_response("Item added to cart successfully", {
                'cart_id': cart_id,
                'product_id': product_id,
                'quantity': total_quantity,
                'item_total': str(cart.items[product_id].total_price)
            })
    
    def remove_item_from_cart(self, cart_id: str, product_id: str) -> Dict[str, any]:
        """Remove item completely from cart"""
        
        if cart_id not in self.carts:
            return self._error_response("Cart not found")
        
        cart = self.carts[cart_id]
        
        if product_id not in cart.items:
            return self._error_response("Item not in cart")
        
        with self.lock:
            # Release inventory reservation
            quantity = cart.items[product_id].quantity
            self.inventory_manager.release_inventory(product_id, quantity)
            
            # Remove item
            del cart.items[product_id]
            cart.update_timestamp()
            
            return self._success_response("Item removed from cart", {
                'cart_id': cart_id,
                'product_id': product_id,
                'released_quantity': quantity
            })
    
    def update_item_quantity(self, cart_id: str, product_id: str, new_quantity: int) -> Dict[str, any]:
        """Update quantity of item in cart"""
        
        if new_quantity <= 0:
            return self.remove_item_from_cart(cart_id, product_id)
        
        if cart_id not in self.carts:
            return self._error_response("Cart not found")
        
        cart = self.carts[cart_id]
        
        if product_id not in cart.items:
            return self._error_response("Item not in cart")
        
        with self.lock:
            old_quantity = cart.items[product_id].quantity
            
            # Release old reservation
            self.inventory_manager.release_inventory(product_id, old_quantity)
            
            # Check new quantity availability
            if not self.inventory_manager.check_availability(product_id, new_quantity):
                # Re-reserve old quantity
                self.inventory_manager.reserve_inventory(product_id, old_quantity)
                return self._error_response(f"Insufficient inventory for quantity {new_quantity}")
            
            # Reserve new quantity
            if not self.inventory_manager.reserve_inventory(product_id, new_quantity):
                # Re-reserve old quantity
                self.inventory_manager.reserve_inventory(product_id, old_quantity)
                return self._error_response("Failed to update inventory reservation")
            
            # Update cart item
            cart.items[product_id].quantity = new_quantity
            cart.update_timestamp()
            
            return self._success_response("Item quantity updated", {
                'cart_id': cart_id,
                'product_id': product_id,
                'old_quantity': old_quantity,
                'new_quantity': new_quantity,
                'new_total': str(cart.items[product_id].total_price)
            })
    
    def apply_discount_code(self, cart_id: str, discount_code: str) -> Dict[str, any]:
        """Apply discount code to cart"""
        
        if cart_id not in self.carts:
            return self._error_response("Cart not found")
        
        if discount_code not in self.discount_codes:
            return self._error_response("Invalid discount code")
        
        cart = self.carts[cart_id]
        discount = self.discount_codes[discount_code]
        
        if not discount.is_valid():
            return self._error_response("Discount code expired or usage limit exceeded")
        
        with self.lock:
            # Calculate current total to check min order amount
            pricing_info = self.pricing_strategy.calculate_total(cart, discount)
            
            if pricing_info['subtotal'] < discount.min_order_amount:
                return self._error_response(f"Minimum order amount ₹{discount.min_order_amount} required")
            
            cart.applied_discount_code = discount_code
            cart.update_timestamp()
            
            return self._success_response("Discount code applied successfully", {
                'discount_code': discount_code,
                'discount_amount': str(pricing_info['discount_amount']),
                'new_total': str(pricing_info['final_total'])
            })
    
    def get_cart_details(self, cart_id: str) -> Dict[str, any]:
        """Get complete cart details with pricing"""
        
        if cart_id not in self.carts:
            return self._error_response("Cart not found")
        
        cart = self.carts[cart_id]
        
        # Get applied discount if any
        applied_discount = None
        if cart.applied_discount_code:
            applied_discount = self.discount_codes.get(cart.applied_discount_code)
        
        # Calculate pricing
        pricing_info = self.pricing_strategy.calculate_total(cart, applied_discount)
        
        cart_details = {
            'cart_id': cart_id,
            'user_id': cart.user_id,
            'status': cart.status.value,
            'created_at': cart.created_at.isoformat(),
            'updated_at': cart.updated_at.isoformat(),
            'items_count': len(cart.items),
            'total_quantity': sum(item.quantity for item in cart.items.values()),
            'pricing': {
                'subtotal': str(pricing_info['subtotal']),
                'discount_amount': str(pricing_info['discount_amount']),
                'delivery_charges': str(pricing_info['delivery_charges']),
                'tax_amount': str(pricing_info['tax_amount']),
                'final_total': str(pricing_info['final_total'])
            },
            'applied_discount_code': cart.applied_discount_code,
            'items': []
        }
        
        # Add item details
        for product_id, item in cart.items.items():
            cart_details['items'].append({
                'product_id': product_id,
                'product_name': item.product.name,
                'unit_price': str(item.product.price),
                'quantity': item.quantity,
                'total_price': str(item.total_price),
                'added_at': item.added_at.isoformat()
            })
        
        return self._success_response("Cart details retrieved", cart_details)
    
    def clear_cart(self, cart_id: str) -> Dict[str, any]:
        """Clear all items from cart"""
        
        if cart_id not in self.carts:
            return self._error_response("Cart not found")
        
        cart = self.carts[cart_id]
        
        with self.lock:
            # Release all inventory reservations
            for product_id, item in cart.items.items():
                self.inventory_manager.release_inventory(product_id, item.quantity)
            
            # Clear cart
            cart.items.clear()
            cart.applied_discount_code = None
            cart.update_timestamp()
            
            return self._success_response("Cart cleared successfully", {
                'cart_id': cart_id
            })
    
    def checkout_cart(self, cart_id: str) -> Dict[str, any]:
        """Checkout cart (convert reservations to actual purchases)"""
        
        if cart_id not in self.carts:
            return self._error_response("Cart not found")
        
        cart = self.carts[cart_id]
        
        if not cart.items:
            return self._error_response("Cart is empty")
        
        with self.lock:
            # Final inventory check
            for product_id, item in cart.items.items():
                if not self.inventory_manager.check_availability(product_id, item.quantity):
                    return self._error_response(f"Product {product_id} no longer available in required quantity")
            
            # Calculate final pricing
            applied_discount = None
            if cart.applied_discount_code:
                applied_discount = self.discount_codes.get(cart.applied_discount_code)
                if applied_discount:
                    applied_discount.current_usage += 1  # Increment usage
            
            pricing_info = self.pricing_strategy.calculate_total(cart, applied_discount)
            
            # Mark cart as checked out
            cart.status = CartStatus.CHECKED_OUT
            cart.update_timestamp()
            
            # In real system, would integrate with payment gateway and order management
            order_id = f"ORDER_{uuid.uuid4().hex[:8].upper()}"
            
            return self._success_response("Checkout successful", {
                'order_id': order_id,
                'cart_id': cart_id,
                'final_total': str(pricing_info['final_total']),
                'items_purchased': len(cart.items),
                'checkout_time': datetime.now().isoformat()
            })
    
    def _get_available_quantity(self, product_id: str) -> int:
        """Get available quantity for product"""
        total_inventory = self.inventory_manager.inventory.get(product_id, 0)
        reserved = self.inventory_manager.reserved.get(product_id, 0)
        return max(0, total_inventory - reserved)
    
    def _success_response(self, message: str, data: any = None) -> Dict[str, any]:
        """Create success response"""
        response = {
            'success': True,
            'message': message,
            'timestamp': datetime.now().isoformat()
        }
        if data:
            response['data'] = data
        return response
    
    def _error_response(self, message: str) -> Dict[str, any]:
        """Create error response"""
        return {
            'success': False,
            'error': message,
            'timestamp': datetime.now().isoformat()
        }

class FlashSaleOrchestrator:
    def __init__(self):
        self.redis_client = RedisCluster()
        self.inventory_service = InventoryService()
        self.payment_service = PaymentService()
        self.queue_manager = VirtualQueueManager()
        
    async def handle_flash_sale_request(self, user_id: str, product_id: str, 
                                      quantity: int) -> Dict:
        """
        Handle flash sale purchase request with Indian market optimizations
        """
        try:
            # Step 1: Add user to virtual queue
            queue_position = await self.queue_manager.add_to_queue(
                user_id, product_id, self._get_user_region(user_id)
            )
            
            if queue_position > 1000:  # Queue too long
                return {
                    'status': 'queued',
                    'queue_position': queue_position,
                    'estimated_wait_minutes': queue_position // 100,
                    'message': 'You are in queue for this flash sale'
                }
            
            # Step 2: Check inventory with atomic operations
            inventory_check = await self.inventory_service.atomic_check_and_reserve(
                product_id, quantity, user_id, timeout_seconds=300
            )
            
            if not inventory_check['success']:
                return {
                    'status': 'sold_out',
                    'message': 'Product is sold out'
                }
            
            # Step 3: Process payment with Indian payment methods
            payment_result = await self.payment_service.process_flash_sale_payment(
                user_id, inventory_check['reservation_id'], 
                inventory_check['total_amount']
            )
            
            if payment_result['success']:
                # Confirm order
                order_id = await self._create_flash_sale_order(
                    user_id, product_id, quantity, inventory_check['reservation_id']
                )
                
                return {
                    'status': 'success',
                    'order_id': order_id,

                }
